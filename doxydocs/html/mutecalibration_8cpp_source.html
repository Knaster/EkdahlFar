48eCalibrationResult CalibrateMute::findMuteLevels() {
49 bool debugReport = debugPrintEnabled[debugPrintType::Debug];
50 debugPrintEnabled[debugPrintType::Debug] = false;
51
52 debugPrintln("Starting the mute level calibration", debugPrintType::TextInfo);
53
54 // Home mute and set mute and bow position at zero and turn off bow
55 muteControl->home();
56
57// audioFilterBiquad->setLowpass(0, bowControl->calibrationDataConnect->fundamentalFrequency, 0.707);
58 bowControl->disableBowMotorPower();
59 bowControl->setRun(false);
60 bowControl->setBowRest();
61 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
62
63 // Find the pickup output level that is considered silence
64 delay(3000);
65 levelSilence = findLevel();
66 if (levelSilence == 0) { levelSilence = 0.01; }
67
68 debugPrintln("Silence level is set to " + String(levelSilence), debugPrintType::TextInfo);
69 if (levelSilence >= (minFundamentalAmplitude / 2)) { return CR_Err_AudioTooHigh; }
70
71 // Enable bow and play the fundamental, set the test pressure to in between max & min pressure
72
73 bowControl->setPIDOn(true);
74 bowControl->setRun(true);
75 bowControl->enableBowMotorPower();
76 harmonicSeriesHandler->setHarmonic(0);
77
78 uint16_t testPressure = bowControl->getEngagePressure() +
79 ((bowControl->getMaxPressure() - bowControl->getEngagePressure()) / 4);
80
81 debugPrintln("Setting tilt to " + String(testPressure), debugPrintType::TextInfo);
82 bowControl->setHardwarePressure(testPressure);
83 debugPrintln("Waiting for Pressure setting to finish.", debugPrintType::TextInfo);
84 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
85 debugPrintln("Pause for a little..", debugPrintType::TextInfo);
86 delay(2500);
87
88 debugPrintln("Starting fundamental level test, need to reach " + String(minFundamentalAmplitude), debugPrintType::TextInfo);
89 // Wait for the bow to settle and then reord the fundamental level
90 do {
91 testPressure += 100;
92 bowControl->setHardwarePressure(testPressure);
93 levelFundamental = findLevel();
94 debugPrintln("Fundamental level " + String(levelFundamental), debugPrintType::TextInfo);
95 } while((levelFundamental < minFundamentalAmplitude) && (testPressure < bowControl->getMaxPressure()));
96
97 if (levelFundamental < minFundamentalAmplitude) { return CR_Err_AudioTooLow; }
98
99 debugPrintln("Pickup fundamental ampltiude is " + String(levelFundamental), debugPrintType::TextInfo);
100
101 #define levelStepSize 100
102
103 float level = 0;
104 uint16_t mutePos = 0;
105 muteControl->setHardwarePosition(mutePos);
106 if (!muteControl->completeTask(5000)) { return CR_Err_MutePressure; }
107
108 do {
109 muteControl->setHardwarePosition(mutePos);
110 if (!muteControl->completeTask(100)) { return CR_Err_MutePressure; }
111
112 mutePos += levelStepSize;
113 level = findLevel();
114
115 debugPrintln("Level " + String(level) + " at position " + String(mutePos), debugPrintType::TextInfo);
116 if (level < minAmplitude) { break; }
117 if (level < levelSilence) { break; }
118 } while ((mutePos < stallPosition) && ((level >= minAmplitude) || (level >= levelSilence)));
119
120 if (mutePos == levelStepSize) { return CR_Err_Undefined; }
121
122 muteControl->setFullMutePosition(mutePos);
123 muteControl->setHalfMutePosition(mutePos / 2);
124 debugPrintln("Full mute set at " + String(muteControl->getFullMutePosition()), debugPrintType::TextInfo);
125
126 debugPrintln("Finished level set", debugPrintType::TextInfo);
127
128 bowControl->setRun(false);
129 bowControl->setBowRest();
130 bowControl->disableBowMotorPower();
131 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
132
133 debugPrintEnabled[debugPrintType::Debug] = debugReport;
134
135 return CR_Ok;
136}

138eCalibrationResult CalibrateMute::findMuteStall() {
139 bool debugReport = debugPrintEnabled[debugPrintType::Debug];
140 debugPrintEnabled[debugPrintType::Debug] = false;
141
142 debugPrintln("Starting mute stall calibration", debugPrintType::TextInfo);
143
144 muteControl->home();
145 bool autoCorrect = muteControl->getAutoCorrect();
146 muteControl->setAutoCorrect(false);
147
148 muteControl->setHardwarePosition(maxTestStep);
149 if (!muteControl->completeTask()) { return CR_Err_MutePressure; }
150 bool initialHomingSensed = muteControl->getHomingSensed();
151
152 debugPrintln("Starting at max with homing sense " + String(initialHomingSensed), debugPrintType::TextInfo);
153
154 uint16_t i = maxTestStep;
155 muteControl->setHardwarePosition(i);
156 if (!muteControl->completeTask()) { return CR_Err_MutePressure; }
157
158 do {
159 i--;
160 muteControl->setHardwarePosition(i);
161 if (!muteControl->completeTask()) { return CR_Err_MutePressure; }
162
163 if (i == 0) {
164 debugPrintln("Edge detection failed with edge " + String(muteControl->getHomingSensed()), debugPrintType::Error);
165 return CR_Err_Undefined;
166 }
167 } while (muteControl->getHomingSensed() == initialHomingSensed);
168
169 debugPrintln("Homing sense changed to " + String(muteControl->getHomingSensed()) + " direction " + String(muteControl->getMoveDirection()) + " step no " +
170 String(muteControl->getCurrentStep()), debugPrintType::TextInfo);
171
172 eEdgeType edge;
173 if (initialHomingSensed == homeSenseInactive) { edge = eEdgeType::EDGERISING; } else { edge =eEdgeType::EDGEFALLING; };
174 uint16_t homingPoint = muteControl->getHomingPoint(edge, eStepDirection::REVERSE);
175 int32_t homingDifference = muteControl->getCurrentStep() - homingPoint;
176 float mult = 65535 / (stepsPerRevolution * microSteps);
177 int32_t maxPosition = maxTestStep - (homingDifference * mult) - maxStepRetract;
178
179 stallPosition = maxPosition;
180
181 debugPrintln("Homing point is at " + String(homingPoint) + " difference is " + String(homingDifference) + " making the max position " + String(maxPosition), debugPrintType::TextInfo);
182
183 muteControl->setAutoCorrect(autoCorrect);
184 debugPrintEnabled[debugPrintType::Debug] = debugReport;
185 return CR_Ok;
186}
