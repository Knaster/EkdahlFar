57bool calibrateMute::findMuteLevels() {
58 bool debugReport = debugPrintEnabled[debugPrintType::Debug];
59 debugPrintEnabled[debugPrintType::Debug] = false;
60
61 debugPrintln("Starting the mute level calibration", debugPrintType::TextInfo);
62
63 // Home mute and set mute and bow position at zero and turn off bow
64 m_muteConnect->homeMute();
65// servoStepper *ss = m_muteConnect->stepServoStepper;
66 servoStepper *ss = m_muteConnect->tmc2209ServoStepper->stepServoStepper;
67
68// audioFilterBiquad->setLowpass(0, m_bowControlConnect->calibrationDataConnect->fundamentalFrequency, 0.707);
69
70 m_bowIOConnect->disableBowPower();
71 m_bowControlConnect->run = 0;
72 m_bowControlConnect->bowRest(1);
73 m_bowIOConnect->tmc2209ServoStepper->stepServoStepper->completeTask();
74
75 // Find the pickup output level that is considered silence
76
77 delay(3000);
78 levelSilence = findLevel();
79 if (levelSilence == 0) { levelSilence = 0.01; }
80
81 debugPrintln("Silence level is set to " + String(levelSilence), debugPrintType::TextInfo);
82 if (levelSilence >= (minFundamentalAmplitude / 2)) {
83 debugPrintln("Silence level is too high!", debugPrintType::Error);
84 return false;
85 }
86
87 // Enable bow and play the fundamental, set the test pressure to in between max & min pressure
88
89 m_bowControlConnect->PIDon = true;
90 m_bowControlConnect->run = 1;
91 m_bowIOConnect->enableBowPower();
92 m_bowControlConnect->setHarmonic(0);
93
94 uint16_t testPressure = m_bowControlConnect->calibrationDataConnect->firstTouchPressure +
95 ((m_bowControlConnect->calibrationDataConnect->stallPressure - m_bowControlConnect->calibrationDataConnect->firstTouchPressure) / 4);
96 debugPrintln("Setting tilt to " + String(testPressure), debugPrintType::TextInfo);
97 m_bowIOConnect->setTiltPWM(testPressure);
98 debugPrintln("Waiting for Pressure setting to finish.", debugPrintType::TextInfo);
99 m_bowIOConnect->waitForTiltToComplete();
100 debugPrintln("Pause for a little..", debugPrintType::TextInfo);
101 delay(2500);
102
103 debugPrintln("Starting fundamental level test, need to reach " + String(minFundamentalAmplitude), debugPrintType::TextInfo);
104 // Wait for the bow to settle and then reord the fundamental level
105 do {
106 testPressure += 100;
107 m_bowIOConnect->setTiltPWM(testPressure);
108 levelFundamental = findLevel();
109 debugPrintln("Fundamental level " + String(levelFundamental), debugPrintType::TextInfo);
110 } while((levelFundamental < minFundamentalAmplitude) && (testPressure < m_bowControlConnect->calibrationDataConnect->stallPressure));
111
112 if (levelFundamental < minFundamentalAmplitude) {
113 debugPrintln("Pickup volume too low!", debugPrintType::Error);
114 return false;
115 }
116
117 debugPrintln("Pickup fundamental ampltiude is " + String(levelFundamental), debugPrintType::TextInfo);
118
119 #define levelStepSize 100
120
121 float level = 0;
122 uint16_t mutePos = 0;
123 do {
124 ss->setPosition(mutePos);
125 ss->completeTask();
126 mutePos += levelStepSize;
127 level = findLevel();
128
129 debugPrintln("Level " + String(level) + " at position " + String(mutePos), debugPrintType::TextInfo);
130 if (level < minAmplitude) { break; }
131 if (level < levelSilence) { break; }
132 } while ((mutePos < stallPosition) && ((level >= minAmplitude) || (level >= levelSilence)));
133
134 if (mutePos == levelStepSize) {
135 debugPrintln("Error finding mute level", debugPrintType::Error);
136 return false;
137 }
138
139 m_muteConnect->setFullMutePosition(mutePos);
140 m_muteConnect->setHalfMutePosition(mutePos / 2);
141 debugPrintln("Full mute set at " + String(m_muteConnect->getFullMutePosition()), debugPrintType::TextInfo);
142
143 debugPrintln("Finished level set", debugPrintType::TextInfo);
144
145 m_bowControlConnect->run = 0;
146 m_bowControlConnect->bowRest(1);
147 m_bowIOConnect->disableBowPower();
148 m_bowIOConnect->tmc2209ServoStepper->stepServoStepper->completeTask();
149
150// audioFilterBiquad->setLowpass(0, noteFreqCutoff, 0.707);
151 debugPrintEnabled[debugPrintType::Debug] = debugReport;
152
153 return true;
154}

156bool calibrateMute::findMuteStall() {
157 bool debugReport = debugPrintEnabled[debugPrintType::Debug];
158 debugPrintEnabled[debugPrintType::Debug] = false;
159
160 debugPrintln("Starting mute stall calibration", debugPrintType::TextInfo);
161
162 m_muteConnect->homeMute();
163// servoStepper *ss = m_muteConnect->stepServoStepper;
164 servoStepper *ss = m_muteConnect->tmc2209ServoStepper->stepServoStepper;
165
166/*
167 debugPrintln("Mute home positions: rising, forward " +
168 String(ss->homingPoint[ss->eEdgeType::EDGERISING][ss->eStepDirection::FORWARD]) +
169 ", falling forward " + String(ss->homingPoint[ss->eEdgeType::EDGEFALLING][ss->eStepDirection::FORWARD]) +
170 ", rising reverse " + String(ss->homingPoint[ss->eEdgeType::EDGERISING][ss->eStepDirection::REVERSE]) +
171 ", falling reverse " + String(ss->homingPoint[ss->eEdgeType::EDGEFALLING][ss->eStepDirection::REVERSE]),
172 debugPrintType::TextInfo);
173*/
174 bool autoCorrect = ss->autoCorrectPosition;
175 ss->autoCorrectPosition = false;
176
177 ss->setPosition(maxTestStep);
178 if (!ss->completeTask()) {
179 debugPrintln("Failed at going to max pressure", debugPrintType::TextInfo);
180 return false;
181 }
182 bool initialHomingSensed = ss->getHomingSensed();
183
184 debugPrintln("Starting at max with homing sense " + String(initialHomingSensed), debugPrintType::TextInfo);
185
186 uint16_t i = maxTestStep;
187 do {
188 i--;
189 ss->setPosition(i);
190 ss->completeTask();
191 if (i == 0) {
192 debugPrintln("Edge detection failed with edge " + String(ss->getHomingSensed()), debugPrintType::Error);
193 return false;
194 }
195 } while (ss->getHomingSensed() == initialHomingSensed);
196
197
198 debugPrintln("Homing sense changed to " + String(ss->getHomingSensed()) + " direction " + String(ss->getMoveDirection()) + " step no " + String(ss->getCurrentStep()), debugPrintType::TextInfo);
199
200 servoStepper::eEdgeType edge;
201 if (initialHomingSensed == homeSenseInactive) { edge = ss->eEdgeType::EDGERISING; } else { edge = ss->eEdgeType::EDGEFALLING; };
202 uint16_t homingPoint = ss->getHomingPoint(edge, ss->eStepDirection::REVERSE);
203 int32_t homingDifference = ss->getCurrentStep() - homingPoint;
204 float mult = 65535 / (stepsPerRevolution * microSteps);
205 int32_t maxPosition = maxTestStep - (homingDifference * mult) - maxStepRetract;
206
207 stallPosition = maxPosition;
208
209 debugPrintln("Homing point is at " + String(homingPoint) + " difference is " + String(homingDifference) + " making the max position " + String(maxPosition), debugPrintType::TextInfo);
210/*
211 debugPrintln("Setting max position for test", debugPrintType::TextInfo);
212 ss->setPosition(maxPosition);
213 ss->completeTask();
214*/
215 ss->autoCorrectPosition = autoCorrect;
216 debugPrintEnabled[debugPrintType::Debug] = debugReport;
217 return true;
218}
