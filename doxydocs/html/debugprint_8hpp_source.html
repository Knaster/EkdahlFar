19enum debugPrintType { Command, USB, Hardware, Undefined, Priority, Error, InfoRequest, EParser, Debug, TextInfo, Help};
20#define debugPrintTypes 11
21String debugPrintTypeName[debugPrintTypes] = { "command", "usb", "hardware" , "undefined", "priority", "error", "inforequest", "expressionparser", "debug", "textinfo", "help" };
22String debugPrintTypeNameShort[debugPrintTypes] = { "cmd", "usb", "hw", "un", "pri", "err", "irq", "ep", "dbg", "txi", "hlp"};
23bool debugPrintEnabled[debugPrintTypes] = { true, true, true, true, true, true, true, false, true, true, true };

25bool debugPrintCheckType(debugPrintType printType) {
26 if (debugPrintEnabled[printType] == true) { return true; }
27 return false;
28}
29void debugPrint(String text, debugPrintType printType) {
30 if (debugPrintCheckType(printType)) { ssOutput.print("[" + debugPrintTypeNameShort[printType] + "]" + text); }
31 return;
32}

43bool setDebugReporting(String debugName, bool setting) {
44 for (int i=0; i<debugPrintTypes; i++) {
45 if (debugPrintTypeName[i] == debugName) {
46 debugPrintEnabled[i] = setting;
47 debugPrintln("Setting reporting of name " + debugName + " to " + setting, Command);
