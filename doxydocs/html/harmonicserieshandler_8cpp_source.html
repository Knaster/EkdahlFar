256bool HarmonicSeriesHandler::updateHarmonicData() {
257 ModuleGroup *group = getGroup("harmonicseries");
258 if (group == nullptr) { return false; }
259
260 HarmonicSeries *module = group->getSingleSelection();
261 if (module == nullptr) { return false; }
262
263 int targetHarmonic = pHarmonic + pHarmonicAdd;
264
265 int harmonicCount = module->ratios.size();
266 if (harmonicCount == 0) {
267 debugPrintln("Harmonic list empty!", debugPrintType::Error);
268 return false;
269 }
270 // Calculate where in the series the current harmonic resides (0-11)
271 int series = targetHarmonic % harmonicCount;
272 if (series < 0) { series = harmonicCount + series; }
273 // If harmonic is below 0 we need to reduce for the truncation to work properly
274 if (targetHarmonic < 0) { targetHarmonic -= (harmonicCount - 1); }
275 int octave = trunc(targetHarmonic / harmonicCount);
276
277 float freq = fundamentalFrequency * pow(2, octave) * module->getHarmonic(series); // - 0.4;
278
279 currentHarmonicFreq = freq;
280 calculateHarmonicShift();
281
282 return raiseFrequencyChanged();
283}
