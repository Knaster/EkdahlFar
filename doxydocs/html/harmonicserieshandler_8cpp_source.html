294bool HarmonicSeriesHandler::updateHarmonicData() {
295 int targetHarmonic = harmonic + harmonicAdd;
296
297 int harmonicCount = harmonicSeriesList.series[currentHarmonicSeries].ratio.size();
298 if (harmonicCount == 0) {
299 debugPrintln("Harmonic list empty!", debugPrintType::Error);
300 return false;
301 }
302 // Calculate where in the series the current harmonic resides (0-11)
303 int series = targetHarmonic % harmonicCount;
304 if (series < 0) { series = harmonicCount + series; }
305 // If harmonic is below 0 we need to reduce for the truncation to work properly
306 if (targetHarmonic < 0) { targetHarmonic -= (harmonicCount - 1); }
307 int octave = trunc(targetHarmonic / harmonicCount);
308
309 float freq = fundamentalFrequency * pow(2, octave) * harmonicSeriesList.series[currentHarmonicSeries].ratio[series]; // - 0.4;
310
311 currentHarmonicFreq = freq;
312 calculateHarmonicShift();
313
314 return raiseFrequencyChanged();
315}
