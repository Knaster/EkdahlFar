37bool bowIO::setBowMotorVoltage(float voltage) {
38 bowMotorVoltage = voltage;
39 voltage -= DCDCLOWERBOUND;
40 float t = (65536 / (DCDCUPPERBOUND - DCDCLOWERBOUND)) * voltage;
41 uint16_t pwm = 65536 - ((uint16_t) t);
42 analogWrite(bowMotorVoltagePin, pwm);
43 return true;

141 if ((freq >= tachoFreqPerimssibleMAX) || (freq <= tachoFreqPerimssibleMIN)) {
142 freq = 0;

146 tachoFreqIndex++;

172float bowIO::averageFreq() {
173 if (tachoFreqCount < tachoFreqLength) {
174 if (tachoFreq[tachoFreqIndex] == 0) {
175 return 0;
176 } else {
177 debugPrintln("AverageFreq: error, not enough point stored.", Priority);
178 return -1;
179 }
180 }
181
182 float localBuffer[tachoFreqLength];
183 for (int i=0; i<tachoFreqLength; i++) { localBuffer[i] = tachoFreq[i]; }
184
185 // Sort the buffer so that middle will be median
186 std::sort(localBuffer, localBuffer + tachoFreqLength);
187 // Take out the median value
188 float median = localBuffer[tachoFreqLength / 2];
189
190 float average = 0;
191 int averageCount = tachoFreqLength;
192
193 for (int i=0; i < tachoFreqLength; i++) {
194 if ((localBuffer[i] > (median + permissibleFreqDeviation)) || (localBuffer[i] < (median - permissibleFreqDeviation))) {
195 averageCount--;
196 } else {
197 average += localBuffer[i];
198 }
199 }
200
201 average /= averageCount;
202 // oldAverage = average; // moved to below
204 if ((average < 0) || (average > 1000)) {
205 debugPrint("AverageFreq: error, data out of range " + String(average) + " with data points " + String(averageCount), Error);
206 debugPrint("Series ", Error);
207 for (int i=0; i<tachoFreqLength; i++) { debugPrint(String(localBuffer[i]) + ", ", Error); }
208 debugPrintln(".", Error);
209 //average = -1;
210 average = tachoOldAverage;
211 }
212 tachoOldAverage = average; // was above before, check if problem
213
214 return average;
215}

218void bowIO::setSpeedPWMSafe(uint16_t speed) {
219 int a = speed;
220 // ramp the speed if going up, driver doesn't like sudden fast changes
221 analogWrite(bowMotorRevPin, 65535 - a);
222 if (lastBowMotorPWM < a) {
223 for (int i=lastBowMotorPWM; i<a; i++) {
224 delayMicroseconds(5); //5
225 analogWrite(bowMotorRevPin, 65535 - i);
226 };
227 } else {
228 for (int i=lastBowMotorPWM; i>a; i--) {
229 delayMicroseconds(5); //5
230 analogWrite(bowMotorRevPin, 65535 - i);
231 };
232 }
233
234 lastBowMotorPWM = a;
235}

270void bowIO::tachoISRHandler() {
271 asm("nop");
272 uint8_t state = digitalRead(reflectorInterruptPin);
273 if (state == lastReflectorISRState) { return; }
274 if (state == 0) {
275 reflectorCyclePeriod = reflectorCounter;
276 reflectorCounter = 0;
277 }
278 addTachoFreq(((double) 1000000 / reflectorCyclePeriod));
279 asm("dsb");
280 lastReflectorISRState = state;
281}

388bool bowIO::bowOverPower() {
389 if (getBowCurrent() >= (bowMotorWattage / bowMotorVoltage)) {
390 // If we have already set the internal over-power flag
391 if (transientOverPower) {
392 // And that was set more than the allowed time-span ago, aka the over-power event has been going on for X ms
393 if (lastBowOverPowerEvent > bowOverPowerDuration) {
394 // Signal the external over power flag
395 bowOverPowerFlag = true;
396 }
397 } else {
398 // If this is a new over-power event set the flag and clear the event-time
399 transientOverPower = true;
400 lastBowOverPowerEvent = 0;
401 }
402 return true;
403 } else {
404 // Clear everything if we no longer have an over power event
405 bowOverPowerFlag = false;
406 transientOverPower = false;
407 return false;
408 }
409}
