37bool bowIO::setBowMotorVoltage(float voltage) {

39 voltage -= DCDCLOWERBOUND;

172float bowIO::averageFreq() {
173 if (tachoFreqCount < tachoFreqLength) {
174 //return -1;
175 if (tachoFreq[tachoFreqIndex] == 0) {
176 return 0;
177 } else {
178 debugPrintln("AverageFreq: error, not enough point stored.", Priority);
179 //return tachoOldAverage;
180 return -1;
181 }
182 }
183
184 float localBuffer[tachoFreqLength];
185 for (int i=0; i<tachoFreqLength; i++) { localBuffer[i] = tachoFreq[i]; }
186
187 // Sort the buffer so that middle will be median
188 std::sort(localBuffer, localBuffer + tachoFreqLength);
189 // Take out the median value
190 float median = localBuffer[tachoFreqLength / 2];
191
192 float average = 0;
193 int averageCount = tachoFreqLength;
194
195 for (int i=0; i < tachoFreqLength; i++) {
196 if ((localBuffer[i] > (median + permissibleFreqDeviation)) || (localBuffer[i] < (median - permissibleFreqDeviation))) {
197 averageCount--;
198 } else {
199 average += localBuffer[i];
200 }
201 }
202
203 average /= averageCount;
204 // oldAverage = average; // moved to below
206 if ((average < 0) || (average > 1000)) {
207 debugPrint("AverageFreq: error, data out of range " + String(average) + " with data points " + String(averageCount), Error);
208 debugPrint("Series ", Error);
209 for (int i=0; i<tachoFreqLength; i++) { debugPrint(String(localBuffer[i]) + ", ", Error); }
210 debugPrintln(".", Error);
211 //average = -1;
212 average = tachoOldAverage;
213 }
214 tachoOldAverage = average; // was above before, check if problem
215
216 return average;
217}

221 int a = speed;
222 // digitalWrite(ZXBMFwdPin, 1);

224 analogWrite(bowMotorRevPin, 65535 - a);
225 if (lastBowMotorPWM < a) {

227 delayMicroseconds(5); //5
228 analogWrite(bowMotorRevPin, 65535 - i);
229 };
230 } else {
231 for (int i=lastBowMotorPWM; i>a; i--) {
232 delayMicroseconds(5); //5
233 analogWrite(bowMotorRevPin, 65535 - i);
234 };
235 }
236
237 lastBowMotorPWM = a;

301void bowIO::tachoISRHandler() {
302 asm("nop");
303 //char state = digitalRead(reflectorInterruptPin);
304 uint8_t state = digitalRead(reflectorInterruptPin);
305 if (state == lastReflectorISRState) { return; }
306 if (state == 0) {
307 reflectorCyclePeriod = reflectorCounter;
308 reflectorCounter = 0;
309 }
310 addTachoFreq(((double) 1000000 / reflectorCyclePeriod));
311 asm("dsb");
312 lastReflectorISRState = state;
313}

443bool bowIO::bowOverPower() {
444 if (getBowCurrent() >= (bowMotorWattage / bowMotorVoltage)) {
445 // If we have already set the internal over-power flag
446 if (transientOverPower) {
447 // And that was set more than the allowed time-span ago, aka the over-power event has been going on for X ms
448 if (lastBowOverPowerEvent > bowOverPowerDuration) {
449 // Signal the external over power flag
450 bowOverPowerFlag = true;
451 }
452 } else {
453 // If this is a new over-power event set the flag and clear the event-time
454 transientOverPower = true;
455 lastBowOverPowerEvent = 0;
456 }
457 return true;
458 } else {
459 // Clear everything if we no longer have an over power event
460 bowOverPowerFlag = false;
461 transientOverPower = false;
462 return false;
463 }
464}
