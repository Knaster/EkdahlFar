215 void calculateAcceleration() {
216 int stepsFromEnd = 0;
217// int stepsFromStart = 0;
218
219 if (moveDirection == eStepDirection::FORWARD) {
220 stepsFromEnd = movingTo - currentStep;
221 } else {
222 stepsFromEnd = currentStep - movingTo;
223 }
224
225 if (stepsFromEnd <= accelerationSteps) {
226 currentAccelStep--;
227 } else {
228 currentAccelStep++;
229 }
230
231 if (currentAccelStep < 0) { currentAccelStep = 0; }
232 if (currentAccelStep > accelerationSteps) { currentAccelStep = accelerationSteps; }
233
234 uAccelPeriod = speedUPeriod + speedUPeriod * (accelerationSteps - currentAccelStep) * accelerationMultiplier;
235 }

414// if (outputDebugData) { debugPrintln("setPosition: moving the same direction", debugPrintType::Debug); }

416// if (outputDebugData) { debugPrintln("setPosition: Is in turn around, you have to do something here!!", debugPrintType::Debug); }

420// if (outputDebugData) { debugPrintln("setPosition: moving the opposite direction", debugPrintType::Debug); }

422// if (outputDebugData) { debugPrintln("setPosition: Is in turn around, you have to do something here!!", debugPrintType::Debug); }
423 }
424 turnAroundPosition = wantToMoveTo;
425 turnAroundDirection = getDirectionTo(wantToMoveTo);
426
427 if ((!turnAround) && ((currentStep + accelerationSteps) > movingTo)) {
428// if (outputDebugData) { debugPrintln("setPosition: Odd thing here, maybe at fault?", debugPrintType::Debug); }
429
430 if (moveDirection == eStepDirection::FORWARD) {
431 movingTo = currentStep + accelerationSteps;
432 } else {
433 movingTo = currentStep - accelerationSteps;
434 }
435 }
436 turnAround = true;
437 inhibitInterrupt = false;
438 return true;
439 }
440 } else {
441 currentAccelStep = 0;
442 }
443 movingTo = wantToMoveTo;
444
445 setDirection(getDirectionTo(movingTo));
446 isMoving = true;
447// debugPrintln("Setting reached target to false", debugPrintType::Debug);
448 reachedTarget = false;
449 updatePosition(false);
450 inhibitInterrupt = false;

463 /***** HOMING COMMANDS ONLY PAST THIS POINT ******
464
465
466 Homing sequence:
467
468 1) if homingswitch is not active, go to 2.1
469 1.1) if homingswitch is active go in the opposite direction oh homingDirection until the homingswitch is inactive and then continue to 1
470 2) move in homingDirection until homingswitch is homeSenseActive
471 2.1) if the amount of steps surpass that of homingStepsThreshold, go in the opposite direction of homingDirection until the homingswitch is active and then
472 continue to 1.1
473 3) record position as homingPoint:RISING,homingDirection = homingOffset (currentStep = homingOffset)
474 4) move forward in homingDirection until homingswitch is homeSenseInactive
475 5) record position as homingPoint:FALLING,homingDirection = currentStep
476 6) move forward past this point seconHomeOverStep number of steps
477 7) move backwards in !homingDirection until homingSwitch is homeSenseActive
478 8) record position as homingPoint:RISING,!homingDirection = currentStep
479 9) move backwards in !homingDirection until homingSwitch is homeSenseInActive
480 10) record position as homingPoint:FALLING,!homingDirection = currentStep
481 11) move to offset position
482 */
