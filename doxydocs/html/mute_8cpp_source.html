25mute::mute(char stepEnPin, char stepDirPin, char stepStepPin, HardwareSerial *stepSerialPort, char stepHomeSensorPin) {
26 stepSerialStream = stepSerialPort;
27 stepServoStepper = new servoStepper(stepStepPin, stepDirPin, stepHomeSensorPin); //, stepEnPin);
28 stepTMC2209Driver = new TMC2209();
29 setupTMC2209();
30 stepServoStepper->setHomingOffset(6000);
31 debugPrintln("Starting home", Debug);
32 stepTMC2209Driver->setRunCurrent(stepHomeCurrentPercent);
33 stepServoStepper->home(servoStepper::eStepDirection::FORWARD,5,5,servoStepper::eStepDirection::REVERSE);
34 if (stepServoStepper->completeTask(5000)) {
35 debugPrintln("Homed", Debug);
36 } else {
37 debugPrintln("Homing FAILED!", Error);
38 }
39 stepTMC2209Driver->setRunCurrent(stepRunCurrentPercent);
40 stepServoStepper->setPosition(0);
41 stepServoStepper->completeTask();
42 stepServoStepper->setSpeed(15);
43}

45bool mute::setupTMC2209() {
46 if (stepTMC2209Driver == nullptr) { return false; }
47
48 stepTMC2209Driver->setup(*stepSerialStream);
49 delay(stepConnectDelay);
50
51 pinMode(stepEnPin, OUTPUT);
52 digitalWrite(stepEnPin, 0);
53
54 stepTMC2209Driver->setRunCurrent(stepRunCurrentPercent);
55 stepTMC2209Driver->setMicrostepsPerStep(stepMicrostepping);
56 stepTMC2209Driver->disableStealthChop();
57 stepTMC2209Driver->setHoldCurrent(10);
58 stepTMC2209Driver->enable();
59 return true;
60}

231bool mute::setSustain(bool inSustain) {
232 // If sustain is changed after keys are released, update mute position
233 if ((inSustain == false) && (sustain == true)) {
234 if (mutePosition == emutePosition::mpFull) {
235 hwMutePos = fullMutePosition;
236 }
237 }
238
239 if ((inSustain == true) && (sustain == false)) {
240 hwMutePos = restPosition;
241 }
242
243 sustain = inSustain;
244 return true;
245}

274bool mute::homeMute(bool invert = false) {
275 debugPrintln("Starting home", Debug);
276
277 stepServoStepper->setHomingOffset(6000);
278 stepTMC2209Driver->setRunCurrent(stepHomeCurrentPercent);
279 stepServoStepper->eStop();
280 if (!invert) {
281 stepServoStepper->home(servoStepper::eStepDirection::FORWARD,5,5,servoStepper::eStepDirection::REVERSE); // ,5,5
