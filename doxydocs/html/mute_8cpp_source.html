34bool Mute::setPosition(uint16_t position) {
35 uint32_t a = restPosition + ((double)(fullMutePosition - restPosition) / 65535 * ((double)position));
36
37 if (a < 0) { a = 0; }
38 if (a > 65535) { a = 65535; }
39 if (a > fullMutePosition) { a = fullMutePosition; }
40 debugPrintln("Setting mute tilt to " + String(a) + ", full mute position " + String(fullMutePosition), debugPrintType::Hardware);
41
42 lastPosition = a;
43 mutePosition = emutePosition::mpUndefined;
44 hwMutePos = a;
45 return true;
46}

105bool Mute::setSustain(bool inSustain) {
106 // If sustain is changed after keys are released, update mute position
107 if ((inSustain == false) && (sustain == true)) {
108 if (mutePosition == emutePosition::mpFull) {
109 hwMutePos = fullMutePosition;
110 }
111 }
112
113 if ((inSustain == true) && (sustain == false)) {
114 hwMutePos = restPosition;
115 }
116
117 sustain = inSustain;
118 return true;
119}

125void Mute::updateMute() {
126 if (hwMutePos != hwLastMutePos) {
127 tmc2209ServoStepper->stepServoStepper->setPosition(hwMutePos);
128 hwLastMutePos = hwMutePos;
129 }
130
131 if ((backOffTime > 0) && (mutePosition == mpFull) && (backOffTimer > backOffTime) && (!tmc2209ServoStepper->stepServoStepper->getIsMoving())) {
132 debugPrintln("Backing off mute", debugPrintType::Debug);
133 rest();
134 }
135
136 if (tmc2209ServoStepper->stepServoStepper->intDrivenMsgFlag) {
137 debugRaw(tmc2209ServoStepper->stepServoStepper->intDrivenMsg);
138 tmc2209ServoStepper->stepServoStepper->intDrivenMsgFlag = false;
139 }
140}
