138bool mute::setTilt(uint16_t tilt) {
139 uint32_t a = restPosition + ((double)(fullMutePosition - restPosition) / 65535 * ((double)tilt));
140
141 if (a < 0) { a = 0; }
142 if (a > 65535) { a = 65535; }
143 if (a > fullMutePosition) { a = fullMutePosition; }
144 debugPrintln("Setting mute tilt to " + String(a) + ", full mute position " + String(fullMutePosition), debugPrintType::Hardware);
145
146 lastTilt = a;
147 mutePosition = emutePosition::mpUndefined;
148 hwMutePos = a;
149 return true;
150}

221bool mute::setSustain(bool inSustain) {
222 // If sustain is changed after keys are released, update mute position
223 if ((inSustain == false) && (sustain == true)) {
224 if (mutePosition == emutePosition::mpFull) {
225 hwMutePos = fullMutePosition;
226 }
227 }
228
229 if ((inSustain == true) && (sustain == false)) {
230 hwMutePos = restPosition;
231 }
232
233 sustain = inSustain;
234 return true;
235}
