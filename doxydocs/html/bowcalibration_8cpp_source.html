43bool CalibrateBow::waitForBowToStabilize(uint16_t maxIterations = 10) {
44 int average;
45 int iterations = 0;
46 int newAverage = bowControl->getAverageTachometerFreq();
47 do {
48 average = newAverage;
49 delay(250);
50 bowControl->checkTachometerTimeout();
51 newAverage = bowControl->getAverageTachometerFreq();
52 iterations++;
53 if (iterations > maxIterations) {
54 debugPrintln("Bow did not stabilize within the given iterations", Priority);
55 return false;
56 }
57 } while (average != newAverage);
58 return true;
59}

64eCalibrationResult CalibrateBow::findMinPressure() {
65 debugPrintln("Finding min contact pressure for minimum usable speed", TextInfo);
66
67 uint16_t minPWM = bowControl->getBowMotorMinPWM();
68 if (minPWM <= 0) { return CR_Err_MotorNotCalibrated; }
69 bowControl->setBowMotorDirectPWM(minPWM);
70
71 bowControl->setHardwarePressure(0);
72 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
73 if (!waitForBowToStabilize()) { return CR_Err_BowStabilize; }
74 delay(500);
75
76 float initialMotorFreq = 0;
77 int tempTimeOut = 0;
78 do {
79 initialMotorFreq = bowControl->getAverageTachometerFreq();
80 delayMicroseconds(10);
81 tempTimeOut++;
82 } while((initialMotorFreq <= 0) && (tempTimeOut<100));
83 if (initialMotorFreq <= 0) { return CR_Err_MotorFrequencyReading; }
84 debugPrintln("Starting freq " + String(initialMotorFreq) + String("Hz"), TextInfo);
85
86 uint16_t i;
87 int average;
88 for (i =0; i<maxTestPressure; i+=10) {
89 bowControl->setHardwarePressure(i);
90 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
91
92 do {
93 average = bowControl->getAverageTachometerFreq();
94 } while (average <= 0);
95
96 if (bowControl->getAverageTachometerFreq() < (initialMotorFreq - deviation)) {
97 break;
98 }
99 }
100 if (bowControl->getAverageTachometerFreq() <= 0) { return CR_Err_MotorFrequencyReading; }
101 debugPrintln("intial contact found at " + String(i) + " frequency " + String(bowControl->getAverageTachometerFreq()) + "Hz", TextInfo);
102
103 if (i > pressureTestRetract) {
104 i -= pressureTestRetract;
105 } else {
106 i = 0;
107 }
108
109 bowControl->setBowMotorDirectPWM(0);
110 bowControl->setHardwarePressure(i);
111 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
112 debugPrintln("starting over at PWM " + String(i), TextInfo);
113
114 bowControl->setBowMotorDirectPWM(bowControl->getBowMotorMinPWM());
115 if (!waitForBowToStabilize()) { return CR_Err_BowStabilize; }
116
117 bool fault = false;
118
119 bowControl->setHardwarePressure(i);
120 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
121
122 do {
123 bowControl->setHardwarePressure(i);
124 if (!bowControl->waitForPressureToSettle(1)) {
125 fault = true;
126 break;
127 }
128
129 delayMicroseconds(100);
130 if (bowControl->getAverageTachometerFreq() < (initialMotorFreq - deviation)) {
131 break;
132 }
133 i++;
134 } while(1 != 2);
135 if ((bowControl->getAverageTachometerFreq() <= 0) || (fault)) { return CR_Err_MotorFrequencyReading; }
136
137 bowControl->setEngagePressure(i);
138
139 debugPrintln("Final contact PWM " + String(i), TextInfo);
140 debugPrintln(" at frequency " + String(bowControl->getAverageTachometerFreq()) + "Hz", TextInfo);
141
142 //bowControl->setBowMotorDirectPWM(0);
143 //bowControl->setHardwarePressure(0);
144
145 return safeExitWithResult(CR_Ok);
146}

151eCalibrationResult CalibrateBow::findMaxPressure() {
152 debugPrintln("Finding max contact", TextInfo);
153
154 bowControl->setBowMotorDirectPWM(bowControl->getBowMotorMinPWM() + (bowControl->getBowMotorMaxPWM() - bowControl->getBowMotorMinPWM()) / 2);
155 bowControl->setHardwarePressure(0);
156 if (!bowControl->waitForPressureToSettle()) { return CR_Err_BowPressure; }
157
158 delay(500);
159 if (!waitForBowToStabilize(100)) { return CR_Err_BowStabilize; }
160
161 float initialMotorFreq = bowControl->getAverageTachometerFreq();
162 if (initialMotorFreq == 0) { return CR_Err_MotorFrequencyReading; }
163
164 int16_t j = bowControl->getEngagePressure();
165
166 if (j > pressureTestRetract) {
167 j -= pressureTestRetract;
168 } else {
169 j = 0;
170 }
171
172 eCalibrationResult calibrationResult = CR_Ok;
173
174 bowControl->setHardwarePressure(j);
175 if (bowControl->waitForPressureToSettle(5000)) {
176 debugPrintln("Starting freq " + String(initialMotorFreq) + "Hz", TextInfo);
177
178 uint16_t i;
179 float average;
180 bool fault = false;
181
182 for (i =j; i<maxTestPressure; i+=1) {
183 bowControl->setHardwarePressure(i);
184 if (!bowControl->waitForPressureToSettle(5000)) { return CR_Err_BowPressure; }
185
186 delayMicroseconds(100);
187 average = bowControl->getAverageTachometerFreq();
188 bowControl->checkTachometerTimeout();
189
190 if ((average == 0) || bowControl->bowMotorIsOverPower() || bowControl->bowMotorIsOverCurrent()) {
191 fault = true;
192 break;
193 }
194 }
195
196 bowControl->setMaxPressure(i);
197 if (!fault) {
198 debugPrintln("Final contact PWM " + String(i) + " at frequency " + String(bowControl->getAverageTachometerFreq()) + "Hz", TextInfo);
199 } else {
200 calibrationResult = CR_Err_MotorFault;
201 }
202 } else {
203 calibrationResult = CR_Err_BowPressure;
204 }
205
206 bowControl->setBowMotorDirectPWM(0);
207 bowControl->setHardwarePressure(0);
208 delay(1000);
209 bowControl->home();
210
211 return safeExitWithResult(calibrationResult);
212}

214eCalibrationResult CalibrateBow::findMinMaxPressure() {
215 bool autoCorrectPositionSave = bowControl->getPressureAutocorrect();
216 bowControl->setPressureAutocorrect(false);
217
218 bowControl->setSpeedMode(eSpeedMode::Manual);
219 bool PIDOn = bowControl->getPIDOn();
220 bowControl->setPIDOn(false);
221 bowControl->enableBowMotorPower();
222
223 float speed = bowControl->getPressureMotorSpeed();
224 bowControl->setPressureMotorSpeed(5);
225
226 eCalibrationResult result = findMinPressure();
227 if (result == CR_Ok) { result = findMaxPressure(); }
228
229 if (result == CR_Ok) {
230 if (bowControl->getEngagePressure() > 3000) {
231 bowControl->setEngagePressure(bowControl->getEngagePressure() - 3000);
232 } else {
233 bowControl->setEngagePressure(0);
234 }
235
236 if (bowControl->getEngagePressure() > 3000) {
237 bowControl->setRestPressure(bowControl->getEngagePressure() - 3000);
238 } else {
239 bowControl->setRestPressure(0);
240 }
241 debugPrintln("Good results acquired, setting firstTouchPressure to " + String(bowControl->getEngagePressure()) + " and restPosition to " +
242 String(bowControl->getRestPressure()), debugPrintType::Debug);
243 }/* else {
244 debugPrintln("Results invalid, not modifying data" , debugPrintType::Error);
245 }*/
246
247 bowControl->setPressureAutocorrect(autoCorrectPositionSave);
248 bowControl->setPressureMotorSpeed(speed);
249
250 return safeExitWithResult(result, PIDOn);
251}

273eCalibrationResult CalibrateBow::findMinMaxSpeedPWM() {
274 int speed = 0;
275 uint32_t i =0;
276 float avgFreq = 0;
277 eCalibrationResult result = CR_Ok;
278
279 float bowMotorPowerLimit = bowControl->getBowMotorMaxPower();
280
281 bool PIDOn = bowControl->getPIDOn();
282 bowControl->setPIDOn(false);
283 bowControl->enableBowMotorPower();
284
285 debugPrintln("Finding min speed", TextInfo);
286 bowControl->setBowMotorDirectPWM(0);
287 bowControl->setHardwarePressure(i);
288 if (!bowControl->waitForPressureToSettle()) { return safeExitWithResult(CR_Err_BowPressure, PIDOn); }
289
290 // Go from 0 PWM to 65535 and break the first time that the frequency is above zero
291 do {
292 bowControl->setBowMotorDirectPWM(speed);
293 delayMicroseconds(1000);
294 speed++;
295 } while ((bowControl->getAverageTachometerFreq() < minPermissibleBowSpeedForCalibration) && (speed < 65536));
296
297
298 delay(500);
299 if (!waitForBowToStabilize() || (speed == 65536)) { return safeExitWithResult(CR_Err_BowStabilize, PIDOn); }
300
301 bowControl->setBowMotorMinPWM(speed);
302 delay(1000);
303 bowControl->setBowMotorMinHz(bowControl->getAverageTachometerFreq());
304 debugPrintln("Min speed is " + String(bowControl->getBowMotorMinHz()) + "Hz given at PWM " + String(bowControl->getBowMotorMinPWM()), TextInfo);
305
306 debugPrintln("Finding max speed", TextInfo);
307
308 bowControl->setBowMotorMaxPower(bowControl->getBowMotorMaxPower() * maxPowerUseMultiplierDuringCalibration);
309
310 debugPrintln("Setting calibration max power to " + String(bowControl->getBowMotorMaxPower()) + " watts", TextInfo);
311 do {
312 bowControl->setBowMotorDirectPWM(i);
313 delayMicroseconds(100);
314 i+=10;
315 //debugPrintln("Motor power use " + String(bowControl->getBowMotorPower()) + " watts at PWM " + String(i), TextInfo);
316
317 } while ((i < 65535) && (!bowControl->bowMotorIsOverPower()));
318 if (i > 65535) { i = 65535; }
319
320 bowControl->setBowMotorMaxPower(bowMotorPowerLimit);
321
322 bowControl->setBowMotorMaxPWM(i);
323 if (i < 65540) { debugPrintln("Motor current limited at " + String(bowControl->getBowMotorMaxPower()) + " watts", TextInfo); }
324 delay(500);
325 if (!waitForBowToStabilize()) { return safeExitWithResult(CR_Err_BowStabilize, PIDOn); }
326
327 avgFreq = bowControl->getAverageTachometerFreq();
328
329 if (avgFreq < minPermissibleBowSpeedForCalibration) { return safeExitWithResult(CR_Err_MotorFrequencyReading, PIDOn); }
330 bowControl->setBowMotorMaxHz(avgFreq);
331
332 debugPrintln("Max speed " + String(bowControl->getBowMotorMaxHz()) + "Hz given at PWM " + String(bowControl->getBowMotorMaxPWM()), TextInfo);
333
334 return safeExitWithResult(result);
