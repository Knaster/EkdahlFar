31eProcessResult Solenoid::processSerialCommand(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request, bool delegate, commandList *delegatedCommands) {
32
33 processCommandItems(inCommandItem, serialCommandsSolenoid, sizeof(serialCommandsSolenoid) / sizeof(serialCommandItem));
34
35 if (inCommandItem->command == "help") {
36 addCommandHelp(serialCommandsSolenoid, sizeof(serialCommandsSolenoid) / sizeof(serialCommandItem), commandResponses,"");
37 return eProcessResult::PassThrough;
38 } else
39 if (inCommandItem->command == "solenoidengage") {
40 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
41 solenoidEngage(inCommandItem->argument[0].toInt());
42 commandResponses->push_back({"se:" + String(inCommandItem->argument[0].toInt()), InfoRequest});
43 } else
44 if (inCommandItem->command == "solenoiddisengage") {
45 solenoidDisengage();
46 commandResponses->push_back({"sd:" + String(inCommandItem->argument[0].toInt()), InfoRequest});
47 } else
48 if (inCommandItem->command == "solenoidengageduration") {
49 if (request) {
50 commandResponses->push_back({ "sed:" + String(solenoidEngageDuration), InfoRequest });
51 } else {
52 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
53 solenoidEngageDuration = inCommandItem->argument[0].toInt();
54 commandResponses->push_back({"sed:" + String(solenoidEngageDuration), InfoRequest});
55 }
56 } else
57 if (inCommandItem->command == "solenoidmaxforce") {
58 if (request) {
59 commandResponses->push_back({ "sxf:" + String(forceMax), InfoRequest });
60 } else {
61 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
62 setSolenoidMax(inCommandItem->argument[0].toInt());
63 commandResponses->push_back({"sfx:" + String(forceMax), InfoRequest});
64 }
65 } else
66 if (inCommandItem->command == "solenoidminforce") {
67 if (request) {
68 commandResponses->push_back({ "sif:" + String(forceMin), InfoRequest });
69 } else {
70 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
71 setSolenoidMin(inCommandItem->argument[0].toInt());
72 commandResponses->push_back({"sif:" + String(forceMin), InfoRequest});
73 }
74 } else
75 if (inCommandItem->command == "solenoidforcemultiplier") {
76 if (request) {
77 commandResponses->push_back({ "sfm:" + String(forceMultiplier), InfoRequest });
78 } else {
79 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
80 setSolenoidMultiplier(inCommandItem->argument[0].toFloat());
81 commandResponses->push_back({"sfm:" + String(forceMultiplier), InfoRequest});
82 }
83 } else {
84 return eProcessResult::NotFound;
85 }
86 return eProcessResult::Ok;
87}

95void Solenoid::solenoidEngage(int force) {
96 if (force <= 0) { return; }
97 if (force > 65535) { force = 65535; }
98
99 float forceMultiplied = ((float) force) * forceMultiplier;
100 if (forceMultiplied == 0) { return; }
101
102 float actualForce = forceMin + ((float) (forceMax - forceMin)) / 65535 * forceMultiplied;
103
104 solenoidEngageTime = micros();
105
106 analogWrite(solenoidPin, int(actualForce));
107 solenoidEngaged = true;
108 debugPrintln("Egaging solenoid with force " + String(actualForce), Hardware);
109}

156bool Solenoid::update() {
157 unsigned long currentTime = micros();
158 if ((solenoidEngaged) && (currentTime - solenoidEngageTime > solenoidEngageDuration) && (solenoidEngageDuration != 0)) {
159 solenoidDisengage();
160 return true;
161 }
162 return false;
163}
