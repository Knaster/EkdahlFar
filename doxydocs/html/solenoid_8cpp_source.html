31 solenoid(char _solenoidPin) {
32 solenoidPin = _solenoidPin;
33 pinMode(solenoidPin, OUTPUT);
34 //digitalWrite(solenoidPin, 1);
35 solenoidPWM = new Teensy_PWM(solenoidPin, 20000, 0);
36 }

38 bool solenoidEngaged = false;
39 unsigned long solenoidEngageTime = 0;
40 unsigned long solenoidEngageDuration = 15000;
41
42 uint16_t forceMax = 65535;
43 uint16_t forceMin = 0;
44 float forceMultiplier = 1;
45
47 void solenoidEngage(int force) {
48 if (force <= 0) { return; }
49 if (force > 65535) { force = 65535; }
50
51 float forceMultiplied = ((float) force) * forceMultiplier;
52// debugPrintln("forceMultiplied " + String(forceMultiplied) + " forceMultiplier " + String(forceMultiplier) + " force " + String(force), Debug);
53 if (forceMultiplied == 0) { return; }
54
55 float actualForce = forceMin + ((float) (forceMax - forceMin)) / 65535 * forceMultiplied;
56
57 solenoidEngageTime = micros();
58
59 // analogWrite(solenoidPin, force);
60 analogWrite(solenoidPin, int(actualForce));
61 solenoidEngaged = true;
62 debugPrintln("Egaging solenoid with force " + String(actualForce), Hardware);
63 }
64
66 void solenoidEngage() {
67 solenoidEngage(65535);
68 }
69
71 void solenoidDisengage() {
72 analogWrite(solenoidPin, 0);
73 debugPrintln("Disengaging solenoid", Hardware);
74 solenoidEngaged = false;
75 }
76
77 void solenoidDisengage(int value) {
78 if (value > 0 ) { solenoidDisengage(); }
79 }
80
81 bool updateSolenoid() {
82 unsigned long currentTime = micros();
83 if ((solenoidEngaged) && (currentTime - solenoidEngageTime > solenoidEngageDuration) && (solenoidEngageDuration != 0)) {
84 solenoidDisengage();
85 return true;
86 }
87 return false;
88 }
89
90 String dumpData() {
91 String dump = "";
92 dump += "sxf:" + String(forceMax) + ",";
93 dump += "sif:" + String(forceMin) + ",";
94 return dump;
95 }
96
97};
