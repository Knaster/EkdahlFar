412void processSerialCommands() {
413 if (ssReader.read()) { commands->addCommands(ssReader.c_str()); }
414
415 if (commands->item.size() == 0) { return; }
416
417 std::vector <commandResponse> commandResponses;
418
419 uint16_t i = 0;
420 eProcessResult processResult;
421
422 while (i < commands->item.size()) {
423 commandItem *tCommandItem = &commands->item[i];
424 bool request = false;
425
426 if ((tCommandItem->command == "requestinfo") || (tCommandItem->command == "rqi")) {
427 if (checkArgumentsMin(tCommandItem, &commandResponses, 1)) {
428 tCommandItem->command = tCommandItem->argument[0].trim();
429 tCommandItem->argument.erase(tCommandItem->argument.begin());
430 } else {
431 debugPrintln("Error converting request", debugPrintType::Error);
432 }
433 request = true;
434 }
435 processCommandItems(tCommandItem, serialCommandsMain, sizeof(serialCommandsMain) / sizeof(serialCommandItem));
436
437 processResult = eProcessResult::NotFound;
438
439 if (tCommandItem->command == "help") {
440 String help = ":\"Below is a list of avaliable commands given in both short and long form, either works. The number and type(s) of arguments are listed afterwards and a brief description. Command usage: command:argument1:argument2 [etc]\"";
441 commandResponses.push_back({help, debugPrintType::Help});
442 addCommandHelp(serialCommandsMain, sizeof(serialCommandsMain) / sizeof(serialCommandItem), &commandResponses,"");
443 processResult = eProcessResult::PassThrough;
444 } else
445 if (tCommandItem->command == "helphidden") {
446 String help = ":\"These commands are hidden for a reason and should not be used unless you REALLY know what you're doing\"";
447 commandResponses.push_back({help, debugPrintType::Help});
448 processResult = eProcessResult::PassThrough;
449 } else {
450 processResult = processMainCommands(tCommandItem, &commandResponses, request);
451 }
452
453 if ((processResult == eProcessResult::NotFound) || (processResult == eProcessResult::PassThrough)) {
454 processResult = farSingle->processSerialCommand(tCommandItem, &commandResponses, request);
455 }
456
457 if ((processResult == eProcessResult::NotFound) || (processResult == eProcessResult::PassThrough)) {
458 processResult = farSingle->processSerialCommandHidden(tCommandItem, &commandResponses, request);
459 }
460
461 if ((processResult == eProcessResult::NotFound) && (tCommandItem->command != "help") && (tCommandItem->command != "helpHidden")) {
462 commandResponses.push_back({"Command not found -" + tCommandItem->command + "-", debugPrintType::Error});
463 }
464
465 i++;
466 }
467 commands->item.clear();
468
469 for (i=0; i<commandResponses.size(); i++) {
470 debugPrintln(commandResponses[i].response, commandResponses[i].responseType);
471 }
472}
