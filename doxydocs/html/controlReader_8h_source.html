25class averager
26{
27 public:
28 int32_t value = 0; // Final value after averaging
29 uint8_t dataAverageLength = 1; // Amount of data points to store and average
30 uint16_t dataIndex; // Current pointer in data array
31 uint16_t dataArray[10]; // Data array
32 uint16_t dataCount; // Amount of data stored, will be the same as dataAverageLength at all times except startup
33 private:
34 bool pDataChanged = false; // Flag to signal that the data has changed in accordance to the error parameters
35 elapsedMillis timeSinceChange = 0; // Time measuring class for error calculation
36 elapsedMillis tempEM;
37
38 public:
39// uint16_t errorTimeThreshold = 10; //20, 10, 50 // Amount of time that has elapsed since last time the dataChanged flag has been signaled
40 uint16_t continuousTimeout = 10;
41 //uint8_t errorThreshold = 2; //2, 0
42 uint16_t continuousErrorThreshold = 2; //PCB5 test - 5
43// uint8_t delayErrorThreshold = 5; //10, 5, 20 // Amount of data change required to signal dataChanged after errorTimeThreshold has elapsed
44 uint16_t interruptedErrorThreshold = 40; //PCB5 test - 20
45
46 bool trigger = false; // Set to true to enable a pDataChanged whenever the data goes above or below the value given by triggerThreshold
47 uint16_t triggerThreshold = 250; // Threshold which at to trigger a data change
48 int16_t triggerDelay = 3000; // Delay from trigger to sampling, given in uS
49 elapsedMicros triggerDelayEM;
50 bool triggerDelayEngage = false; // Flag to indicate trigger has been received and sampling delay engaged
51
52 bool singleShot = false;
53
54 bool outputDebugData = false;
55
56 averager() {
57 dataIndex = 0;
58 dataCount = 0;
59
60 for (int i = 0; i < dataAverageLength; i++) {
61 dataArray[i] = 0;
62 }
63 }
64
65 bool dataChanged() {
66 bool temp = pDataChanged;
67 pDataChanged = false;
68 return temp;
69 }
70
71 bool addData(int16_t data) {
72 // If we are still just filling up the buffer then just add the data
73 if ((dataCount + 1) < dataAverageLength) {
74 dataCount++;
75 value = -1;
76
77 dataArray[dataIndex] = data;
78 dataIndex++;
79 if (dataIndex >= dataAverageLength) { dataIndex = 0; }
80 } else {
81// int16_t prevData = dataIndex - 1;
82// if (prevData < 0) { prevData = dataAverageLength - 1; }
83
84 dataArray[dataIndex] = data;
85 dataIndex++;
86 if (dataIndex >= dataAverageLength) { dataIndex = 0; }
87
88 int32_t temp = 0;
89 for (int i = 0; i < dataAverageLength; i++) {
90 temp += dataArray[i];
91 }
92 temp = temp / dataAverageLength;
93
94 if (trigger) {
95 if (triggerDelayEngage) {
96 if (triggerDelayEM >= triggerDelay) {
97 pDataChanged = true;
98 timeSinceChange = 0;
99 triggerDelayEngage = false;
100 }
101 } else
102 if (singleShot) {
103// if (((value <= triggerThreshold) && (temp > triggerThreshold) ))
104 } else
105 if (((value <= triggerThreshold) && (temp > triggerThreshold)) || ((value >= triggerThreshold) && (temp < triggerThreshold))) {
106 debugPrintln("Value " + String(value) + " and average " + String(temp) + " over threshold " + String(triggerThreshold), Debug);
107 if (triggerDelay == 0) {
108 pDataChanged = true;
109 timeSinceChange = 0;
110 } else {
111 triggerDelayEngage = true;
112 triggerDelayEM = 0;
113 }
114 }
115 } else {
116 // If a quick change has happened, aka a new value has been received before the continuousTimeout has been passed
117 // then we are being lenient and passing any value change greater than continuousErrorThreshold. This is to allow continuous sweeps
118 if (timeSinceChange < continuousTimeout) {
119 if ((temp < (value - continuousErrorThreshold)) || (temp > (value + continuousErrorThreshold))) {
120 if (outputDebugData) {
121 debugPrintln("Continuous data change occured at " + String(timeSinceChange) + " ms which is below the continuousTimeout of " + String(continuousTimeout) + " ms."
122 "The data is above the continuousErrorThreshold of " + String(continuousErrorThreshold) + ". The new data is " + String(temp) + " while the old data was " + String(value), Debug);
123 }
124 pDataChanged = true;
125 timeSinceChange = 0;
126 }
127 } else {
128 if ((temp < (value - interruptedErrorThreshold)) || (temp > (value + interruptedErrorThreshold))) {
129 if (outputDebugData) {
130 debugPrintln("A data change occured after the errorThreshold time, the data did surpass the interruptedErrorThreshold of " + String(interruptedErrorThreshold) +
131 ". The new data is " + String(temp) + " while the old data was " + String(value), Debug);
132 }
133 pDataChanged = true;
134 timeSinceChange = 0;
135 }
136 }
137 }
138
139 if (pDataChanged) { value = temp; }
140 }
141 return true;
142 }
143};
