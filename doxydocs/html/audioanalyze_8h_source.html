47void startAudioAnalyze() {
48 audioADC = new AudioInputAnalog(22);
49 audioFilterBiquad = new AudioFilterBiquad();
50 audioFilterBiquad->setLowpass(0, noteFreqCutoff, 0.707);
51#ifndef USB_MIDI_AUDIO_SERIAL
52 audioNoteFreq = new AudioAnalyzeNoteFrequency();
53 audioPeak = new AudioAnalyzePeak();
54 audioRMS = new AudioAnalyzeRMS();
55#endif
56
57#ifdef USB_MIDI_AUDIO_SERIAL
58 audioOutUSB = new AudioOutputUSB();
59#endif
60#ifndef USB_MIDI_AUDIO_SERIAL
61 audioPatchCord3 = new AudioConnection(*audioADC, *audioFilterBiquad);
62 audioPatchCord1 = new AudioConnection(*audioFilterBiquad, *audioNoteFreq);
63 audioPatchCord2 = new AudioConnection(*audioFilterBiquad, *audioPeak);
64 audioPatchCord4 = new AudioConnection(*audioADC, *audioRMS);
65// audioPatchCord1 = new AudioConnection(*audioADC, *audioNoteFreq);
66// audioPatchCord2 = new AudioConnection(*audioADC, *audioPeak);
67#endif
68#ifdef USB_MIDI_AUDIO_SERIAL
69// audioPatchCord3 = new AudioConnection(*audioADC, *audioOutUSB);
70 audioPatchCord3 = new AudioConnection(*audioADC, *audioFilterBiquad);
71 audioPatchCord4 = new AudioConnection(*audioFilterBiquad, *audioOutUSB);
72#endif
73 AudioMemory(30);
74
75#ifndef USB_MIDI_AUDIO_SERIAL
76 audioAnalyzeStarted = true;
77 audioNoteFreq->begin(0.45);
78#endif
79}
