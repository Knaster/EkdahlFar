113 void buildHierarchy() {
114 hierarchyIndex = 0;
115 hierarchy.clear();
116
117 int start = 0, end = 0;
118 String portion;
119 do {
120 end = command.indexOf('.', start);
121 if (end == -1) { end = command.length(); }
122 portion = command.substring(start, end);
123
124 SELECTION thisSel;
125 thisSel.clear();
126
127 //debugPrintln("Portion " + portion, debugPrintType::Debug);
128
129 int i = portion.indexOf('[');
130 int j = portion.lastIndexOf(']');
131
132 if ((i != -1) && (j != -1 )) {
133 i += 1;
134
135 String sel = portion.substring(i, j);
136 //debugPrintln("Adding selection: " + sel, debugPrintType::Debug);
137 int l = 0, m, n, k = 0;
138 while ((k < j)) {
139 k = portion.indexOf(',', i + 1);
140 if (k == -1) { k = j; };
141
142 sel = portion.substring(i, k);
143 l = sel.indexOf('-');
144 if (l == -1) {
145 //debugPrintln("Adding " + sel, debugPrintType::Debug);
146 thisSel.push_back(sel.toInt());
147 } else {
148 m = sel.substring(0, l).toInt();
149 n = sel.substring(l + 1, sel.length()).toInt();
150 if ((n < m) || (m < 0) || (n < 0) || (m > 255) || (n > 255)) {
151 debugPrintln("Invalid selection range", Error);
152 return;
153 }
154 for (m=m; m<=n; m++) {
155 thisSel.push_back(m);
156 //debugPrintln("Adding " + String(m), debugPrintType::Debug);
157 }
158 }
159 i = k + 1;
160 }
161
162 portion = portion.substring(0, portion.indexOf('['));
163 //debugPrintln("New portion is " + portion, debugPrintType::Debug);
164 }
165
166 hierarchy.push_back({ portion, thisSel });
167 start = end + 1;
168 } while (end < command.length());
169
170 }

206 debugPrintln("Found quote end @ " + String(start), debugPrintType::EParser);

210 int parseBracket(String commandString, int start) {
211 debugPrintln("Found bracket start @ " + String(start), debugPrintType::EParser);
212 do {
213 start++;
214 if (String(commandString[start]) == '[') {
215 start = parseBracket(commandString, start);
216 }
217 } while ((String(commandString[start]) != ']') && (start < (int(commandString.length()) - 1)));
218 debugPrintln("Found bracket end @ " + String(start), debugPrintType::EParser);

394String delimitExpression(String expression, bool force) {
395 String delimit = "";
396
397 if (expression == "") { return expression; }
398
399 if (((expression[0] == "'") or (expression[0] == "\"")) and (expression[0] == expression[expression.length() - 1])) {
400// debugPrintln("-" + expression + "- is already delimited!", debugPrintType::Debug);
401 return expression;
402 }
403
404 int singleQ = expression.indexOf("'");
405 int doubleQ = expression.indexOf("\"");
406
407 if (singleQ < doubleQ) {
408 delimit = "'";
409 } else if (singleQ > doubleQ) {
410 delimit = "\"";
411 }
412
413// debugPrintln("exp -" + expression + "-", debugPrintType::Debug);
414 if (!((singleQ == 0) || (doubleQ == 0))) {
415 if (force) {
416// debugPrintln("force", debugPrintType::Debug);

419 delimit = "";
420 }
421 } else {
422// debugPrintln("has quote; singleQ: " + String(singleQ) + ", doubleQ: " + String(doubleQ), debugPrintType::Debug);
423 }
424
425 expression = delimit + expression + delimit;
426// debugPrintln("new exp -" + expression + "-", debugPrintType::Debug);
427
428 if (delimit != "") {
429// debugPrintln("Using quotestyle " + delimit + ", complete expression is " + expression, debugPrintType::Debug);

435String stripQuotes(String expression) {
436 // 1. strip starting and trailing white spaces
437 // 2. strip starting and trailing commas
438 // 3. check if beginning starts with a quote
439 // 3.1 make sure that the end contains the same quote, if not, signal an error
440 // 3.2 strip quotes from start and end
441 expression = expression.trim();
442
443 if ((String(expression[0]) == "\"") || (String(expression[0]) == "'")) {
444// debugPrintln("Stripping start & end quotes from -" + expression + "-", debugPrintType::Debug);
445 if (expression[0] != expression[expression.length()-1]) {
446// debugPrintln("ERROR Quote start and end doesn't match", debugPrintType::Error);
447 }
448 expression = expression.substring(1, expression.length() - 1);
449// debugPrintln("Stripped expression is -" + expression + "-", debugPrintType::Debug);
450 } else {
451// debugPrintln("The first character is -" + String(expression[0]) + "-", debugPrintType::Debug);
452// debugPrintln("No quotes found in the string -" + expression + "-", debugPrintType::Debug);
453 }
454
455 return expression;
456}
