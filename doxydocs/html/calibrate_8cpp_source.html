44bool calibrate::waitForBowToStabilize(uint16_t maxIterations = 10) {
45 int average;
46 int iterations = 0;
47 int newAverage = bowIOConnect->averageFreq();
48 do {
49 average = newAverage;
50 delay(250);
51 bowIOConnect->checkTimeout();
52 newAverage = bowIOConnect->averageFreq();
53 iterations++;
54 if (iterations > maxIterations) {
55 debugPrintln("Bow did not stabilize within the given iterations", Priority);
56 return false;
57 }
58 } while (average != newAverage);
59// debugPrintln("Bow settled at ~" + String(newAverage) + "Hz, last value ~" + String(average) + "Hz", Debug);
60 return true;
61}

66bool calibrate::findMinPressure() {
67 debugPrintln("Finding min contact pressure for minimum usable speed", TextInfo);
68
69 bowIOConnect->setSpeedPWMSafe(calibrationData->minSpeedPWM);
70 bowIOConnect->setTiltPWM(0);
71 bowIOConnect->waitForTiltToComplete();
72 if (!waitForBowToStabilize()) { return false; }
73 delay(500);
74
75 float initialMotorFreq = 0;
76 int tempTimeOut = 0;
77 do {
78 initialMotorFreq = bowIOConnect->averageFreq();
79 delayMicroseconds(10);
80 tempTimeOut++;
81 } while((initialMotorFreq <= 0) && (tempTimeOut<100));
82 if (initialMotorFreq <= 0) { return false; }
83 debugPrintln("Starting freq " + String(initialMotorFreq) + String("Hz"), TextInfo);
84
85 uint16_t i;
86 int average;
87 for (i =0; i<maxTestPressure; i+=10) {
88 bowIOConnect->setTiltPWM(i);
89 bowIOConnect->waitForTiltToComplete(10); // was 3000 (2024-06-27)
90 do {
91 average = bowIOConnect->averageFreq();
92 } while (average <= 0);
93
94 if (bowIOConnect->averageFreq() < (initialMotorFreq - deviation)) {
95 break;
96 }
97 }
98 if (bowIOConnect->averageFreq() <= 0) { return false; }
99 debugPrintln("intial contact found at " + String(i) + " frequency " + String(bowIOConnect->averageFreq()) + "Hz", TextInfo);
100
101 if (i > pressureTestRetract) {
102 i -= pressureTestRetract;
103 } else {
104 i = 0;
105 }
106
107 bowIOConnect->setSpeedPWMSafe(0);
108 bowIOConnect->setTiltPWM(i);
109 bowIOConnect->waitForTiltToComplete();
110 debugPrintln("starting over at PWM " + String(i), TextInfo);
111
112 bowIOConnect->setSpeedPWMSafe(calibrationData->minSpeedPWM);
113 if (!waitForBowToStabilize()) { return false; }
114
115 bool fault = false;
116
117 // Set the tilt PWM with a long delay allowance at first
118 bowIOConnect->setTiltPWM(i);
119 if (!bowIOConnect->waitForTiltToComplete(500)) { // was 100
120 return false;
121 }
122 // Then move on to the test
123 do {
124 bowIOConnect->setTiltPWM(i);
125 if (!bowIOConnect->waitForTiltToComplete(1)) { // was 100
126 fault = true;
127 break;
128 }
129
130 delayMicroseconds(100);
131 if (bowIOConnect->averageFreq() < (initialMotorFreq - deviation)) {
132 break;
133 }
134 i++;
135 } while(1 != 2);
136 if ((bowIOConnect->averageFreq() <= 0) || (fault)) { return false; }
137
138 calibrationData->firstTouchPressure = i;
139
140 debugPrintln("Final contact PWM " + String(calibrationData->firstTouchPressure), TextInfo);
141 debugPrintln(" at frequency " + String(bowIOConnect->averageFreq()) + "Hz", TextInfo);
142
143 bowIOConnect->setSpeedPWMSafe(0);
144 bowIOConnect->setTiltPWM(0);
145
146 return true;
147}

152bool calibrate::findMaxPressure() {
153 debugPrintln("Finding max contact", TextInfo);
154
155// bowControlConnect->run = 1;
156 bowIOConnect->setSpeedPWMSafe( calibrationData->minSpeedPWM + (calibrationData->maxSpeedPWM - calibrationData->minSpeedPWM) / 2 );
157 bowIOConnect->setTiltPWM(0);
158 bowIOConnect->waitForTiltToComplete(5000);
159 delay(500);
160 if (!waitForBowToStabilize(100)) { return false; }
161 float initialMotorFreq = bowIOConnect->averageFreq();
162 if (initialMotorFreq == 0) { return false; }
163
164 int16_t j = calibrationData->firstTouchPressure;
165 if (j > pressureTestRetract) {
166 j -= pressureTestRetract;
167 } else {
168 j = 0;
169 }
170 //bowIOConnect->setTiltPWM((calibrationData->firstTouchPressure - pressureTestRetract));
171 bowIOConnect->setTiltPWM(j);
172 if (bowIOConnect->waitForTiltToComplete(5000)) {
173 debugPrintln("Starting freq " + String(initialMotorFreq) + "Hz", TextInfo);
174
175 uint16_t i;
176 float average;
177 bool fault = false;
178
179// for (i =(calibrationData->firstTouchPressure - pressureTestRetract); i<maxTestPressure; i+=1) {
180 for (i =j; i<maxTestPressure; i+=1) {
181 bowIOConnect->setTiltPWM(i);
182 if (!bowIOConnect->waitForTiltToComplete(100)) {
183 debugPrintln("Tilt didn't finish", Debug);
184 fault = true;
185 break;
186 }
187 delayMicroseconds(100);
188 average = bowIOConnect->averageFreq();
189 bowIOConnect->checkTimeout();
190
191 // debugPrintln("Tilt " + String (i) + " Average frequency " + String(average) + " power " + String(bowIOConnect->getBowCurrent() * bowIOConnect->getBowMotorVoltage())
192 // + ", current " + String(bowIOConnect->getBowCurrent()) + "A, motor fault " + String(bowIOConnect->getMotorFault()), Debug);
193
194 if ((average == 0) || bowIOConnect->bowOverPower() || bowIOConnect->bowOverCurrent()) {
195 break;
196 }
197 }
198
199 calibrationData->stallPressure = i;
200 if (!fault) {
201 debugPrintln("Final contact PWM " + String(calibrationData->stallPressure) + " at frequency " + String(bowIOConnect->averageFreq()) + "Hz"/* with max current " +
202 String(maxCurrentRead)*/, TextInfo);
203 } else {
204 debugPrintln("Got Fault! Final contact PWM " + String(calibrationData->stallPressure) + " at frequency " + String(bowIOConnect->averageFreq()) + "Hz"/* with max current " +
205 String(maxCurrentRead)*/, TextInfo);
206 // debugPrintln("Max pressure calibration ERROR!", debugPrintType::Error);
207 }
208 } else {
209 debugPrintln("Couldn't get calibration start conditions", debugPrintType::Error);
210 }
211
212 bowIOConnect->setSpeedPWMSafe(0);
213 bowIOConnect->setTiltPWM(0);
214 delay(1000);
215 bowIOConnect->homeBow();
216
217 return true;
218}

220bool calibrate::findMinMaxPressure() {
221 // Save the current state of the stepper auto-correction setting and restore before exiting
222 bool autoCorrectPositionSave = bowIOConnect->tmc2209ServoStepper->stepServoStepper->autoCorrectPosition;
223 bowIOConnect->tmc2209ServoStepper->stepServoStepper->autoCorrectPosition = false;
224
225 bool PIDOn = bowControlConnect->PIDon;
226 bowControlConnect->PIDon = 0;
227// bowControlConnect->run = 1;
228 bowIOConnect->enableBowPower();
229
230 uint16_t speed = bowIOConnect->tmc2209ServoStepper->stepServoStepper->getSpeed();
231 bowIOConnect->tmc2209ServoStepper->stepServoStepper->setSpeed(5);
232
233 //bool result = findMinPressure() && findMaxPressure();
234 bool result = findMinPressure();
235 if (result) {
236 result = result & findMaxPressure();
237 }
238
239 if (result) {
240 if (calibrationData->firstTouchPressure > 3000) {
241 calibrationData->firstTouchPressure -= 3000;
242 } else {
243 calibrationData->firstTouchPressure = 0;
244 }
245
246 if (calibrationData->firstTouchPressure > 3000) {
247 calibrationData->restPosition = calibrationData->firstTouchPressure - 3000;
248 } else {
249 calibrationData->restPosition = 0;
250 }
251 debugPrintln("Good results acquired, setting firstTouchPressure to " + String(calibrationData->firstTouchPressure) + " and restPosition to " + String(calibrationData->restPosition), debugPrintType::Debug);
252 } else {
253 debugPrintln("Results invalid, not modifying data" , debugPrintType::Error);
254 }
255
256 bowIOConnect->tmc2209ServoStepper->stepServoStepper->autoCorrectPosition = autoCorrectPositionSave;
257 bowIOConnect->tmc2209ServoStepper->stepServoStepper->setSpeed(speed);
258 bowControlConnect->run = 0;
259 bowControlConnect->PIDon = PIDOn;
260
261 return result;
262}

274bool calibrate::findMinMaxSpeedPWM() {
275// bool finished = false;
276 int speed = 0;
277 uint32_t i =0;
278 float bowMotorWattage = bowIOConnect->bowMotorWattage;
279
280 bool PIDOn = bowControlConnect->PIDon;
281 bowControlConnect->PIDon = 0;
282// bowControlConnect->run = 1;
283 bowIOConnect->enableBowPower();
284
285 debugPrintln("Finding min speed", TextInfo);
286 bowIOConnect->setSpeedPWMSafe(0);
287 bowIOConnect->setTiltPWM(0);
288 bowIOConnect->waitForTiltToComplete();
289
290 // Go from 0 PWM to 65535 and break the first time that the frequency is above zero
291 do {
292 bowIOConnect->setSpeedPWMSafe(speed);
293 delayMicroseconds(1000);
294 speed++;
295 } while ((bowIOConnect->averageFreq() < minPermissibleBowSpeedForCalibration) && (speed < 65536));
296 delay(500);
297 if (!waitForBowToStabilize() || (speed == 65536)) { goto skiptoend; }
298
299 calibrationData->minSpeedPWM = speed;
300 calibrationData->minHz = bowIOConnect->averageFreq();
301 debugPrintln("Min speed " + String(calibrationData->minSpeedPWM) + " at frequency " + String(calibrationData->minHz) + "Hz", TextInfo);
302
303 debugPrintln("Finding max speed", TextInfo);
304
305 bowIOConnect->bowMotorWattage = float(bowIOConnect->bowMotorWattage * maxPowerUseMultiplierDuringCalibration);
306 debugPrintln("Setting calibration max power to " + String(bowIOConnect->bowMotorWattage) + " watts", TextInfo);
307 do {
308 bowIOConnect->setSpeedPWMSafe(i);
309 delayMicroseconds(100);
310 i+=10;
311 debugPrintln("Motor power use " + String(bowIOConnect->getBowCurrent() * bowIOConnect->getBowMotorVoltage()) + " watts at PWM " + String(i), TextInfo);
312 } while ((i < 65535) && (!bowIOConnect->bowOverPower()));
313 if (i > 65535) { i = 65535; }
314 bowIOConnect->bowMotorWattage = bowMotorWattage;
315
316 if (i < 65540) {
317 calibrationData->maxSpeedPWM = i;
318 debugPrintln("Motor current limited at " + String(bowIOConnect->bowMotorWattage * maxPowerUseMultiplierDuringCalibration) + " watts", TextInfo);
319 }
320 delay(500);
321 if (!waitForBowToStabilize()) { goto skiptoend; }
322 if (bowIOConnect->averageFreq() < 1) { goto skiptoend; }
323 calibrationData->maxHz = bowIOConnect->averageFreq();
324 calibrationData->maxSpeedPWM = i;
325
326 // Start at 65535 and decrease PWM until the average frequency is below the recorded max hz, record this as maxSpeedPWM
327
328 debugPrintln("Max speed " + String(calibrationData->maxSpeedPWM) + " at frequency " + String(calibrationData->maxHz) + "Hz", TextInfo);
329
330skiptoend:
331 bowIOConnect->setSpeedPWMSafe(0);
332 bowIOConnect->setTiltPWM(0);
333
334 bowControlConnect->run = 0;
335 bowControlConnect->PIDon = PIDOn;
336
337 return true;

340bool calibrate::findMinMaxSpeedPID() {
341 debugPrintln("Finding min PID speed", TextInfo);
342
343 bowIOConnect->setSpeedPWMSafe(0);
344 bowIOConnect->setTiltPWM(0);
345 bowIOConnect->waitForTiltToComplete();
346
347 float targetFreq = calibrationData->minHz * 3;
348
349 bowIOConnect->enableBowPower();
350 bowControlConnect->setPIDTarget(targetFreq);
351 _speedMode savedSpeedMode = bowControlConnect->speedMode;
352 bowControlConnect->speedMode = _speedMode::Manual;
353 bowControlConnect->run = 1;
354 bowIOConnect->clearTachoData();
355 delay(1500);
356
357 uint16_t testStepDelay = 100;
358
359 // Testing for stability of initial PID stability
360 float readFreq = 0;
361 do {
362 readFreq = 0;
363 for (int i=0; i<25; i++) {
364 delay(1);
365 readFreq += bowIOConnect->averageFreq();
366 }
367 readFreq = (readFreq / 25);
368 debugPrintln("Testing freq " + String(targetFreq) + " read frequency " + String(readFreq), TextInfo);
369
370 if (!((readFreq >= (targetFreq - 1)) && (readFreq <= (targetFreq + 1)))) {
371 break;
372 }
373 targetFreq -= 1;
374
375 bowControlConnect->setPIDTarget(targetFreq);
376 delay(testStepDelay);
377 } while (targetFreq < calibrationData->maxHz);
378
379 calibrationData->minHz = readFreq;
380 debugPrintln("Minimum PID frequency found at " + String(calibrationData->minHz) + " Hz", TextInfo);
381 delay(500);
382
383 bowIOConnect->clearTachoData();
384 targetFreq = calibrationData->maxHz;
385 bowControlConnect->setPIDTarget(targetFreq);
386 // Testing for stability of initial PID stability
387 readFreq = 0;
388 do {
389 readFreq = 0;
390 for (int i=0; i<25; i++) {
391 delay(1);
392 readFreq += bowIOConnect->averageFreq();
393 }
394 readFreq = (readFreq / 25);
395 debugPrintln("Testing freq " + String(targetFreq) + " read frequency " + String(readFreq), TextInfo);
396
397 if ((readFreq >= (targetFreq - 1)) && (readFreq <= (targetFreq + 1))) {
398 break;
399 }
400 targetFreq -= 1;
401
402 bowControlConnect->setPIDTarget(targetFreq);
403 delay(testStepDelay);
404 } while (targetFreq > calibrationData->minHz);
405
406 calibrationData->maxHz = readFreq;
407 debugPrintln("Maximum pid frequency with no pressure found at " + String(calibrationData->maxHz) + " Hz", TextInfo);
408 delay(500);
409
410 bowIOConnect->setTiltPWM(calibrationData->stallPressure);
411 bowIOConnect->waitForTiltToComplete();
412 readFreq = 0;
413 do {
414 readFreq = 0;
415 for (int i=0; i<25; i++) {
416 delay(1);
417 readFreq += bowIOConnect->averageFreq();
418 }
419 readFreq = (readFreq / 25);
420 debugPrintln("Testing freq " + String(targetFreq) + " read frequency " + String(readFreq), TextInfo);
421
422 if ((readFreq >= (targetFreq - 1)) && (readFreq <= (targetFreq + 1))) {
423 break;
424 }
425 targetFreq -= 1;
426
427 bowControlConnect->setPIDTarget(targetFreq);
428 delay(testStepDelay);
429 } while (targetFreq > calibrationData->minHz);
430
431 calibrationData->maxHzPIDFullPressure = readFreq;
432 debugPrintln("Maximum pid frequency with pressure " + String(calibrationData->maxHzPIDFullPressure) + " Hz", TextInfo);
433 delay(500);
434
435 bowControlConnect->speedMode = savedSpeedMode;
436 bowControlConnect->run = 0;
437
438
439 return true;
440}
