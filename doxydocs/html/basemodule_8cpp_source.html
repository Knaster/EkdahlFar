233eProcessResult BaseModule::processCommands(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request) {
234 eProcessResult processResult;
235
236 //debugPrintln("Before processing we have " + String(inCommandItem->argument.size()) + " arguments", debugPrintType::Debug);
237
238 processResult = Module::processCommands(inCommandItem, commandResponses, request);
239 //debugPrintln("After module processing we have " + String(inCommandItem->argument.size()) + " arguments", debugPrintType::Debug);
240 if (processResult == eProcessResult::NotFound) {
241// debugPrintln("Main?", debugPrintType::Debug);
242 if (mainModule != nullptr) {
243// debugPrintln("Main.", debugPrintType::Debug);
244 processResult = mainModule->processCommands(inCommandItem, commandResponses, request);
245 }
246 }
247 //debugPrintln("All commands processed, adding " + String(inCommandItem->argument.size()) + " response arguments", debugPrintType::Debug);
248 String response = inCommandItem->command;
249 for (int i = 0; i < inCommandItem->argument.size(); i++) {
250 response += ":" + inCommandItem->argument[i];
251 }
252 //debugPrintln("All response arguments added", debugPrintType::Debug);
253
254 switch (processResult) {
255 case eProcessResult::NotFound:
256 commandResponses->push_back({"Command not found: " + response, debugPrintType::Error});
257 break;
258 case eProcessResult::CommandFailed:
259 commandResponses->push_back({"Command failed: " + response, debugPrintType::Error});
260 break;
261 case eProcessResult::WrongArgumentCount:
262 case eProcessResult::WrongArgumentMinimum:
263 commandResponses->push_back({"Wrong command arguments: " + response, debugPrintType::Error});
264 break;
265 case eProcessResult::WrongArgumentValue:
266 commandResponses->push_back({"Wrong command argument values: " + response, debugPrintType::Error});
267 break;
268 case eProcessResult::Ok:
269 //debugPrintln("Ok", debugPrintType::TextInfo);
270 break;
271 case eProcessResult::PassThrough:
272 //debugPrintln("Ok, pass-through", debugPrintType::TextInfo);
273 break;
274 default:
275 debugPrintln("Unknown response " + String(processResult), debugPrintType::Error);
276 };
277
278 //debugPrintln("Returning result", debugPrintType::Debug);
279
280 return processResult;
281}
