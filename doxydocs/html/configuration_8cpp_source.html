41uint8_t configuration::setCC(uint8_t controller, String *command) {
42 for (uint16_t i=0; i<controlChange.size(); i++) {
43 if (controlChange[i].control == controller) {
44 controlChange[i].command = *command;
45 return i;
46 }
47 }
48 controlChange.push_back({controller, *command });
49 return controlChange.size() - 1;
50}

52bool configuration::removeCC(uint8_t controller) {
53 for (uint16_t i=0; i<controlChange.size(); i++) {
54 if (controlChange[i].control == controller) {
55 controlChange.erase(controlChange.begin() + i);
56 return true;
57 }
58 }
59 return false;
60}
