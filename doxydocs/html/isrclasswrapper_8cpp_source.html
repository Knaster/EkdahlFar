34bool tachoISR_assignInterrupt(uint8_t interruptPin, bowControl *bowControlObject) {
35 if (tachoISR_bowControlCounter >= tachoISR_bowControlArrayMax) { return false; }
36 tachoISR_bowControlArray[tachoISR_bowControlCounter] = bowControlObject;
37
38 switch(tachoISR_bowControlCounter) {
39 case 0:
40 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR0, CHANGE);
41 debugPrintln("Assigning ISR0", Debug);
42 break;
43 case 1:
44 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR1, CHANGE);
45 break;
46 case 2:
47 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR2, CHANGE);
48 break;
49 case 3:
50 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR3, CHANGE);
51 break;
52 case 4:
53 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR4, CHANGE);
54 break;
55 case 5:
56 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR5, CHANGE);
57 break;
58 case 6:
59 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR6, CHANGE);
60 break;
61 case 7:
62 attachInterrupt(digitalPinToInterrupt(interruptPin), tachoISR_ISR7, CHANGE);
63 break;
64 default:
65 return false;
66}
67tachoISR_bowControlCounter++;
68
69return true;
70}
