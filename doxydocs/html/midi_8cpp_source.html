80 if ((value < (target + epDeadbandThreshold)) && (value > (target - epDeadbandThreshold))) { return target; } else {

141int8_t findNote(byte channel, byte note) {
142 for (int i = 0; i < int(notesHeld.size()); i++) {
143 if ((notesHeld[i].note == note) && (notesHeld[i].channel == channel)) {
144 return i;
145 }
146 }
147 return -1;
148}

150bool removeNote(byte channel, byte note) {
151 int8_t i = findNote(channel, note);
152 if (i == -1) {
153 debugPrintln("Couldn't find note " + String(note), Debug);
154 return false;
155 } else {
156 notesHeld.erase(notesHeld.begin() + i);
157 debugPrintln("Removing note " + String(note), Debug);
158 return true;
159 }
160};

162void addNote(byte channel, byte note, byte velocity) {
163 noteMsg local;
164 local.note = note;
165 local.channel = channel;
166 local.velocity = velocity;
167 if (removeNote(channel, note)) {
168 debugPrintln("Note added twice!!", Error);
169 }
170 notesHeld.push_back(local);
171 debugPrintln("Adding note " + String(note) + " (note no " + String(notesHeld.size()) + ")", Debug);
172};

196void OnNoteOff(byte channel, byte note, byte velocity) {
197 if ((configArray[currentConfig].midiRxChannel != channel) && (configArray[currentConfig].midiRxChannel < 17) &&
198 (configArray[currentConfig].midiRxChannel > 0)) { return; }
199
200 noteCount--;
201 if (noteCount < 0) { noteCount = 0; }
202
203 removeNote(channel, note);
204 dchannel = channel; dnote = note; dvelocity = velocity;
205 processLocalMessage(configArray[currentConfig].noteOff);
206
207 if ((notesHeld.size() > 0) && (monoMode)) {
208 setNotes();
209 }
210
211 debugPrintln("NoteOff with note " + String(note) + " at velocity " + String(velocity), USB);
212}

215void OnNoteOn(byte channel, byte note, byte velocity) {
216 if ((configArray[currentConfig].midiRxChannel != channel) && (configArray[currentConfig].midiRxChannel < 17) &&
217 (configArray[currentConfig].midiRxChannel > 0)) { return; }
218
219 debugPrintln("NoteOn with note " + String(note) + " at velocity " + String(velocity), USB);
220 if (velocity == 0) {
221 OnNoteOff(channel, note, velocity);
222 return;
223 }
224 addNote(channel, note, velocity);
225
226 setNote();
227
228 noteCount++;
229}
