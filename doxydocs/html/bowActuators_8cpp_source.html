38eProcessResult BowActuators::processSerialCommand(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request, bool delegate, commandList *delegatedCommands) {
39
40 processCommandItems(inCommandItem, serialCommandsBowActuators, sizeof(serialCommandsBowActuators) / sizeof(serialCommandItem));
41
42 if (inCommandItem->command == "help") {
43 addCommandHelp(serialCommandsBowActuators, sizeof(serialCommandsBowActuators) / sizeof(serialCommandItem), commandResponses,"");
44 return eProcessResult::PassThrough;
45 } else
46 if (inCommandItem->command == "bowactuator") {
47 if (request) {
48 commandResponses->push_back({ "ba:" + String(m_currentBowActuator), debugPrintType::InfoRequest });
49 } else {
50 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
51 if (!loadBowActuator(inCommandItem->argument[0].toInt())) {
52 return eProcessResult::CommandFailed;
53 }
54 commandResponses->push_back({"ba:" + String(m_currentBowActuator), debugPrintType::InfoRequest});
55 }
56 } else
57 if (inCommandItem->command == "bowactuatorsave") {
58 if (!checkArguments(inCommandItem, commandResponses, 2)) { return eProcessResult::WrongArgumentCount; }
59 if (!saveBowActuator(inCommandItem->argument[0].toInt(), inCommandItem->argument[1])) {
60 return eProcessResult::CommandFailed;
61 }
62 //uint8_t actuator = getBowActuator();
63 commandResponses->push_back({"bas:" + String(m_currentBowActuator) + ":" + getBowActuatorID(m_currentBowActuator), InfoRequest});
64 } else
65 if (inCommandItem->command == "bowactuatordata") {
66 int8_t bowIndex;
67 if (request) {
68 bowIndex = m_currentBowActuator;
69 if (checkArguments(inCommandItem, commandResponses, 1, true)) {
70 bowIndex = inCommandItem->argument[0].toInt();
71 if (bowIndex > (getBowActuatorCount() - 1)) {
72 return eProcessResult::WrongArgumentValue;
73 }
74 } else {
75 return eProcessResult::WrongArgumentCount;
76 }
77 } else {
78 if (!checkArguments(inCommandItem, commandResponses, 5)) { return eProcessResult::WrongArgumentCount; }
79 setBowActuatorData(inCommandItem->argument[0].toInt(), inCommandItem->argument[1].toInt(), inCommandItem->argument[2].toInt(), inCommandItem->argument[3].toInt(), inCommandItem->argument[4]);
80 bowIndex = inCommandItem->argument[0].toInt();
81 }
82 commandResponses->push_back({ "bad:" + String(bowIndex) + ":" + getBowActuatorFirstTouchPressure(bowIndex) + ":" + getBowActuatorStallPressure(bowIndex) + ":" +
83 getBowActuatorRestPosition(bowIndex) + ":" + getBowActuatorID(bowIndex), InfoRequest });
84 } else
85 if (inCommandItem->command == "bowactuatorcount") {
86 commandResponses->push_back({ "bac:" + String(getBowActuatorCount()), InfoRequest });
87 } else
88 if (inCommandItem->command == "bowactuatorremove") {
89 if (!request) {
90 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
91 if (!removeBowActuator(inCommandItem->argument[0].toInt())) { return eProcessResult::CommandFailed; }
92 commandResponses->push_back({"bar:" + String(inCommandItem->argument[0].toInt()), InfoRequest});
93 }
94 } else {
95 return eProcessResult::NotFound;
96 }
97 return eProcessResult::Ok;
98}

121bool BowActuators::removeBowActuator(uint8_t t_actuator) {
122 if (t_actuator >= m_bowActuator.size()) {
123 return false;
124 }
125 m_bowActuator.erase(m_bowActuator.begin() + t_actuator);
126 if (m_currentBowActuator >= t_actuator) {
127 m_currentBowActuator = 0;
128 debugPrintln("Removed active bow actuator, resetting to actuator 0", debugPrintType::Debug);
129 }
130 return true;
131}

171bool BowActuators::saveBowActuator(uint16_t actuator, String name) {
172 if (actuator > m_bowActuator.size() - 1) {
173 debugPrintln("Inserting new actuator", Debug);
174 m_bowActuator.push_back({0, 65535, 0, "new"});
175 actuator = m_bowActuator.size() - 1;
176 }
177 m_bowActuator[actuator].firstTouchPressure = bowPressure->getEngagePressure();
178 m_bowActuator[actuator].stallPressure = bowPressure->getMaxPressure();
179 m_bowActuator[actuator].restPosition = bowPressure->getRestPressure();
180 m_bowActuator[actuator].id = name;
181 return true;
182};
