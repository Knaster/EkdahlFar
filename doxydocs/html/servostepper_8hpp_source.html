189 void calculateAcceleration() {
190 int stepsFromEnd = 0;
191
192 if (moveDirection == eStepDirection::FORWARD) {
193 stepsFromEnd = movingTo - currentStep;
194 } else {
195 stepsFromEnd = currentStep - movingTo;
196 }
197
198 if (stepsFromEnd <= accelerationSteps) {
199 currentAccelStep--;
200 } else {
201 currentAccelStep++;
202 }
203
204 if (currentAccelStep < 0) { currentAccelStep = 0; }
205 if (currentAccelStep > accelerationSteps) { currentAccelStep = accelerationSteps; }
206
207 uAccelPeriod = speedUPeriod + speedUPeriod * (accelerationSteps - currentAccelStep) * accelerationMultiplier;
208 }

297 void updatePositionMove() {
298 if (isMoving) {
299 if (autoCorrectPosition) { correctPosition(); }
300
301 if (((currentStep < movingTo) && (moveDirection == REVERSE)) || ((currentStep > movingTo) && (moveDirection == FORWARD))) {
302 tsDebugPrintln("ERROR! direction, currentStep and movingTo does not match!!", debugPrintType::Error);
303 tsDebugPrintln("updatePositionMove: moving to target " + String(movingTo) + " current step " + String(currentStep) + " direction " + (moveDirection), debugPrintType::Debug);
304 }
305
306 if (currentStep == movingTo) {
307 if ((outputDebugData) && (hasStepped)) { tsDebugPrintln("updatePositionMove: reached target " + String(currentStep), debugPrintType::Debug); }
308
309 if (turnAround) {
310 if (outputDebugData) { tsDebugPrintln("updatePositionMove: reached turnAround", debugPrintType::Debug); }
311 movingTo = turnAroundPosition;
312 setDirection(turnAroundDirection);
313 turnAround = false;
314 } else {
315 reachedTarget = true;
316 isMoving = false;
317 }
318 } else {
319 if (hasStepped) {
320 calculateAcceleration();
321 hasStepped = false;
322 }
323
324 stepNext(uAccelPeriod);
325 if (stepIntervalCallback != nullptr) {
326 stepInterval.begin(stepIntervalCallback, ((double) uAccelPeriod));
327 }
328 }
329 }

332 void correctPosition() {
333 if (homingSensedChanged) {
334 eEdgeType edgeDirection;
335 if (homingSensed) {
336 edgeDirection = eEdgeType::EDGEFALLING;
337 } else {
338 edgeDirection = eEdgeType::EDGERISING;
339 }
340
341 if (currentStep != homingPoint[edgeDirection][moveDirection]) {
342 debugPrintln("Got position " + String(currentStep) + " should have position " + String(homingPoint[edgeDirection][moveDirection]) +
343 " edge " + String(edgeDirection) + " direction " + String(moveDirection) + " movingTo " + String(movingTo) + " - Correcting : stepID " + String(stepperID),
344 debugPrintType::Debug);
345
346 // say we're moving to 1800, the step THINKS it is at 1795 but corrects to 1805 and corrects it to there
347 // then movingTo will miss the target
348
349 if ((moveDirection == eStepDirection::FORWARD) && (homingPoint[edgeDirection][moveDirection] > movingTo)) {
350 debugPrintln("This should become a problem : stepID " + String(stepperID), Error);
351 movingTo = homingPoint[edgeDirection][moveDirection];
352 }
353
354 if ((moveDirection == eStepDirection::REVERSE) && (homingPoint[edgeDirection][moveDirection] < movingTo)) {
355 debugPrintln("This should become a problem : stepID " + String(stepperID), Error);
356 movingTo = homingPoint[edgeDirection][moveDirection];
357 }
358
359 currentStep = homingPoint[edgeDirection][moveDirection];
360 }
361 }

363public:
364 bool setPosition(uint16_t position) {
365 uint16_t wantToMoveTo = ((float) valuePerRotation * position);
366
367 if (outputDebugData) { tsDebugPrintln("setPosition: wantToMoveTo " + String(wantToMoveTo), debugPrintType::Debug); }
368
369 if ((wantToMoveTo == movingTo) && (isMoving)) { return true; }
370
371 inhibitInterrupt = true;
372
373 if (isMoving) {
374 if (moveDirection == getDirectionTo(wantToMoveTo)) {
375 if (turnAround) {
376 debugPrintln("setPosition: Is in turn around, you have to do something here!!", debugPrintType::Debug);
377 }
378 turnAround = false;
379 } else {
380 if (turnAround) {
381 debugPrintln("setPosition: Is in turn around, you have to do something here!!", debugPrintType::Debug);
382 }
383 turnAroundPosition = wantToMoveTo;
384 turnAroundDirection = getDirectionTo(wantToMoveTo);
385
386 if ((!turnAround) && ((currentStep + accelerationSteps) > movingTo)) {
387
388 if (moveDirection == eStepDirection::FORWARD) {
389 movingTo = currentStep + accelerationSteps;
390 } else {
391 movingTo = currentStep - accelerationSteps;
392 }
393 }
394 turnAround = true;
395 inhibitInterrupt = false;
396 return true;
397 }
398 } else {
399 currentAccelStep = 0;
400 }
401 movingTo = wantToMoveTo;
402
403 setDirection(getDirectionTo(movingTo));
404 isMoving = true;
405 reachedTarget = false;
406 updatePosition(false);
407 inhibitInterrupt = false;
408 return true;
409 }

415 void eStop() {
416 isMoving = false;
417 homing = eHomingStage::UNHOMED;
418 }
419
420 /***** HOMING COMMANDS ONLY PAST THIS POINT ******
421
422
423 Homing sequence:
424
425 1) if homingswitch is not active, go to 2.1
426 1.1) if homingswitch is active go in the opposite direction oh homingDirection until the homingswitch is inactive and then continue to 1
427 2) move in homingDirection until homingswitch is homeSenseActive
428 2.1) if the amount of steps surpass that of homingStepsThreshold, go in the opposite direction of homingDirection until the homingswitch is active and then
429 continue to 1.1
430 3) record position as homingPoint:RISING,homingDirection = homingOffset (currentStep = homingOffset)
431 4) move forward in homingDirection until homingswitch is homeSenseInactive
432 5) record position as homingPoint:FALLING,homingDirection = currentStep
433 6) move forward past this point seconHomeOverStep number of steps
434 7) move backwards in !homingDirection until homingSwitch is homeSenseActive
435 8) record position as homingPoint:RISING,!homingDirection = currentStep
436 9) move backwards in !homingDirection until homingSwitch is homeSenseInActive
437 10) record position as homingPoint:FALLING,!homingDirection = currentStep
438 11) move to offset position
439 */

441 void updatePositionFirstHomingRising() {
442 if (abs(currentStep) > homingStepsThreshold) {
443 debugPrintln("Homing past steps threshold, reverting back to switch : stepID " + String(stepperID), debugPrintType::Debug);
444 setDirection(notDirection(homingDirection));
445 homing = eHomingStage::MOVETOHOMESWITCH;
446 }
447
448 if ((homingSensedChanged) && (homingSensed == homeSenseActive)) {
449 if (homingOffsetDirection == eStepDirection::FORWARD) {
450 currentStep = -homingOffset;
451 } else {
452 currentStep = homingOffset;
453 }
454 homingPoint[eEdgeType::EDGERISING][homingDirection] = currentStep;
455 debugPrintln("Setting first homing rising point " + String(homingDirection) + " to " + String(homingPoint[eEdgeType::EDGERISING][homingDirection]) +
456 " stepID " + String(stepperID), debugPrintType::Debug);
457
458 debugPrintln("[firsthomerising] Next direction is " + String(moveDirection) + ", direction inversion is " + String(invertDirection), debugPrintType::Debug);
459
460 homing = eHomingStage::SECONDHOMINGFALLING;
461 } else {
462 stepNext();
463 if (hasStepped) {
464 hasStepped = false;
465 }
466 }

568 bool home(eStepDirection inHomingDirection = eStepDirection::REVERSE, uint8_t inHomingSpeedRough = 10, uint8_t inHomingSpeedFine = 5,
569 eStepDirection inOffsetDirection = eStepDirection::FORWARD) {
570
571 if (isMoving) {
572 debugPrintln("Moving!!! Not homing", debugPrintType::Error);
573 return false;
574 }
575
576 homingDirection = inHomingDirection;
577 homingSpeedRough = inHomingSpeedRough;
578 homingSpeedFine = inHomingSpeedFine;
579 homingOffsetDirection = inOffsetDirection;
580
581 currentStep = 0; // Reset current step
582
583 debugPrintln("Homing direction " + String(homingDirection) + " homing speed r " + String(homingSpeedRough) + " homing speed f " + String(homingSpeedFine) +
584 " homing ofs. dir. " + String(homingOffsetDirection), debugPrintType::Debug);
585
586 setSpeed(homingSpeedRough);
587 if (homingSensed == homeSenseActive) {
588 homing = eHomingStage::MOVEPASTHOMESWITCH;
589 setDirection(notDirection(homingDirection));
590 debugPrintln("On homing switch, moving : stepID " + String(stepperID), debugPrintType::Debug);
591 } else {
592 homing = eHomingStage::FIRSTHOMINGRISING;
593 setDirection(homingDirection);
594 }
595 isMoving = true;
596 return true;
597 }
