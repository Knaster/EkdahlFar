196float DCMotorControl::getAverageTachometerFreq() {
197
198 if (tachoFreqCount < tachoFreqLength) {
199 if (tachoFreq[tachoFreqIndex] == 0) {
200 return 0;
201 } else {
202 debugPrintln("AverageFreq: error, not enough point stored.", Priority);
203 return -1;
204 }
205 }
206
207 float localBuffer[tachoFreqLength];
208 for (int i=0; i<tachoFreqLength; i++) { localBuffer[i] = tachoFreq[i]; }
209
210 // Sort the buffer so that middle will be median
211 std::sort(localBuffer, localBuffer + tachoFreqLength);
212 // Take out the median value
213 float median = localBuffer[tachoFreqLength / 2];
214
215 float average = 0;
216 int averageCount = tachoFreqLength;
217
218 for (int i=0; i < tachoFreqLength; i++) {
219 if ((localBuffer[i] > (median + permissibleFreqDeviation)) || (localBuffer[i] < (median - permissibleFreqDeviation))) {
220 averageCount--;
221 } else {
222 average += localBuffer[i];
223 }
224 }
225
226 average /= averageCount;
227 // oldAverage = average; // moved to below
229 if ((average < 0) || (average > 1000)) {
230 debugPrint("AverageFreq: error, data out of range " + String(average) + " with data points " + String(averageCount), Error);
231 debugPrint("Series ", Error);
232 for (int i=0; i<tachoFreqLength; i++) { debugPrint(String(localBuffer[i]) + ", ", Error); }
233 debugPrintln(".", Error);
234 //average = -1;
235 average = tachoOldAverage;
236 }
237 tachoOldAverage = average; // was above before, check if problem
238
239 return average;
240}

243void DCMotorControl::tachometerISRHandler() {
244 asm("nop");
245 uint8_t state = digitalRead(reflectorInterruptPin);
246 if (state == lastReflectorISRState) { return; }
247 if (state == 0) {
248 reflectorCyclePeriod = reflectorCounter;
249 reflectorCounter = 0;

251 addTachoFreq(((double) 1000000 / reflectorCyclePeriod));
252 asm("dsb");

323bool DCMotorControl::isOverPower() {
324 if (getMotorCurrent() >= (motorPowerLimit / motorVoltage)) {
325 // If we have already set the internal over-power flag
326 if (transientOverPower) {
327 // And that was set more than the allowed time-span ago, aka the over-power event has been going on for X ms
328 if (lastOverPowerEvent > overPowerDuration) {
329 // Signal the external over power flag
330 overPowerFlag = true;
331 }
332 } else {
333 // If this is a new over-power event set the flag and clear the event-time
334 transientOverPower = true;
335 lastOverPowerEvent = 0;
336 }
337 return true;
338 } else {
339 // Clear everything if we no longer have an over power event
340 overPowerFlag = false;
341 transientOverPower = false;
342 return false;
343 }
344}
