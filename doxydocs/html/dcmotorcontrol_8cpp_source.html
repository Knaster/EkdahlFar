98/*
99 * - Calculates the average bow speed frequency using the values stored in the tachometer buffer by
100 * -# finding the median of the buffer
101 * -# recalculating the average skipping values straying from the median by a frequency given by permissibleFreqDeviation
102 *
103 */
104float DCMotorControl::averageFreq() {
105 if (tachoFreqCount < tachoFreqLength) {
106 if (tachoFreq[tachoFreqIndex] == 0) {
107 return 0;
108 } else {
109 debugPrintln("AverageFreq: error, not enough point stored.", Priority);
110 return -1;
111 }
112 }
113
114 float localBuffer[tachoFreqLength];
115 for (int i=0; i<tachoFreqLength; i++) { localBuffer[i] = tachoFreq[i]; }
116
117 // Sort the buffer so that middle will be median
118 std::sort(localBuffer, localBuffer + tachoFreqLength);
119 // Take out the median value
120 float median = localBuffer[tachoFreqLength / 2];
121
122 float average = 0;
123 int averageCount = tachoFreqLength;
124
125 for (int i=0; i < tachoFreqLength; i++) {
126 if ((localBuffer[i] > (median + permissibleFreqDeviation)) || (localBuffer[i] < (median - permissibleFreqDeviation))) {
127 averageCount--;
128 } else {
129 average += localBuffer[i];
130 }
131 }
132
133 average /= averageCount;
134 // oldAverage = average; // moved to below
136 if ((average < 0) || (average > 1000)) {
137 debugPrint("AverageFreq: error, data out of range " + String(average) + " with data points " + String(averageCount), Error);
138 debugPrint("Series ", Error);
139 for (int i=0; i<tachoFreqLength; i++) { debugPrint(String(localBuffer[i]) + ", ", Error); }
140 debugPrintln(".", Error);
141 //average = -1;
142 average = tachoOldAverage;
143 }
144 tachoOldAverage = average; // was above before, check if problem
145
146 return average;
147}

223bool DCMotorControl::overPower() {
224 if (getCurrent() >= (motorWattage / motorVoltage)) {
225 // If we have already set the internal over-power flag
226 if (transientOverPower) {
227 // And that was set more than the allowed time-span ago, aka the over-power event has been going on for X ms
228 if (lastOverPowerEvent > overPowerDuration) {
229 // Signal the external over power flag
230 overPowerFlag = true;
231 }
232 } else {
233 // If this is a new over-power event set the flag and clear the event-time
234 transientOverPower = true;
235 lastOverPowerEvent = 0;
236 }
237 return true;
238 } else {
239 // Clear everything if we no longer have an over power event
240 overPowerFlag = false;
241 transientOverPower = false;
242 return false;
243 }
