57eProcessResult BowControl::processSerialCommand(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request, bool delegate, commandList *delegatedCommands) {
58
59 eProcessResult processResult = eProcessResult::Ok;
60
61 processCommandItems(inCommandItem, serialCommandsBowControl, sizeof(serialCommandsBowControl) / sizeof(serialCommandItem));
62
63 if (inCommandItem->command == "help") {
64 addCommandHelp(serialCommandsBowControl, sizeof(serialCommandsBowControl) / sizeof(serialCommandItem), commandResponses,"");
65 }
66
67 if (((inCommandItem->command == "bowpressurerest") || (inCommandItem->command == "bpr")) && !request) {
68 debugPrintln("Second rest", debugPrintType::Debug);
69 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
70 if (inCommandItem->argument[0].toInt() == 1) { rest(); }
71 }
72
73 if (inCommandItem->command == "bowmotorrun") {
74 if (request) {
75 commandResponses->push_back({ "bmr:" + String(run), InfoRequest });
76 } else {
77 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
78 if (inCommandItem->argument[0].toInt() > 0) {
79 run = 1;
80 dcMotorControl->enableMotorPower();
81 } else {
82 run = 0;
83 dcMotorControl->disableMotorPower();
84 }
85 commandResponses->push_back({"bmr:" + String(run), InfoRequest});
86 }
87 } else
88 if (inCommandItem->command == "bowpid") {
89 if (request) {
90 commandResponses->push_back({ "bpid:" + String(PIDon), InfoRequest });
91 } else {
92 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
93 if (inCommandItem->argument[0].toInt() > 0) { PIDon = true; } else { PIDon = false;}
94 commandResponses->push_back({"bpid:" + String(PIDon), InfoRequest});
95 }
96 } else
97 if (inCommandItem->command == "bowcontrolspeedmode") {
98 if (request) {
99 commandResponses->push_back({"bcsm:" + String(int(speedMode)), InfoRequest});
100 } else {
101 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
102 if (!validateNumber(inCommandItem->argument[0].toInt(), 0, 1)) { return eProcessResult::WrongArgumentValue; }
103 speedMode = (eSpeedMode) (inCommandItem->argument[0].toInt());
104 commandResponses->push_back({"bcsm:" + String(int(speedMode)), InfoRequest});
105 }
106 } else
107 if (inCommandItem->command == "bowmotordirectpwm") {
108 if (request) {
109 commandResponses->push_back({ "bmdp:" + String(manualMotorPWM), InfoRequest });
110 } else {
111 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
112 manualMotorPWM = inCommandItem->argument[0].toInt();
113 commandResponses->push_back({"bmdp:" + String(manualMotorPWM), InfoRequest});
114 }
115 } else
116 if (inCommandItem->command == "bowmotortimeout") {
117 if (request) {
118 commandResponses->push_back({ "bmt:" + String(bowShutoffTimeout), InfoRequest });
119 } else {
120 bowShutoffTimeout = inCommandItem->argument[0].toInt();
121 commandResponses->push_back({"bmt:" + String(bowShutoffTimeout), InfoRequest});
122 }
123 } else
124 if (inCommandItem->command == "bowmotorfaultcommands") {
125 if (!request) {
126 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
127 commandsMotorFault = delimitExpression(inCommandItem->argument[0], true);
128 }
129 commandResponses->push_back({ "bmfc:" + String(commandsMotorFault), InfoRequest });
130 } else
131 if (inCommandItem->command == "bowmotoroverpowercommands") {
132 if (!request) {
133 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
134 commandsOverPowerCurrent = delimitExpression(inCommandItem->argument[0], true);
135 }
136 commandResponses->push_back({ "bmopc:" + String(commandsOverPowerCurrent), InfoRequest });
137 } else {
138 processResult = bowPressure->processSerialCommand(inCommandItem, commandResponses, request, delegate, delegatedCommands);
139 if ((processResult != eProcessResult::NotFound) && (processResult != eProcessResult::PassThrough)) { return processResult; }
140
141 processResult = dcMotorControl->processSerialCommand(inCommandItem, commandResponses, request, delegate, delegatedCommands);
142 if ((processResult != eProcessResult::NotFound) && (processResult != eProcessResult::PassThrough)) { return processResult; }
143
144 processResult = pidController->processSerialCommand(inCommandItem, commandResponses, request, delegate, delegatedCommands);
145 }
146
147 return processResult;
148}

188void BowControl::updateMotorAutoShutdown() {
189 if ((speedMode == eSpeedMode::Automatic) && (bowPressure->getHold() == false) && (bowShutoffTimer >= bowShutoffTimeout)) {
190 if ((bowPressure->getPressureMode() == ePressureMode::Rest) && (bowShutoffTimedout == false)) {
191 run = false;
192 bowShutoffTimedout = true;
193 debugPrintln("Auto shutdown of motor", debugPrintType::Debug);
194 } else
195 if ((bowShutoffTimedout == true) && (bowShutoffMotorDisabled == false) && (dcMotorControl->getAverageTachometerFreq() == 0)) {
196 dcMotorControl->disableMotorPower();
197 bowShutoffMotorDisabled = true;
198 debugPrintln("Auto shutdown of motor dc/dc converter", debugPrintType::Debug);
199 }
200 }
201}

203void BowControl::updateRun_PID() {
204 if (run) {
205 if (PIDon) {
206 if (pidController->getPIDTarget() == 0) { dcMotorControl->setSpeedPWM(0); }
207 } else {
208 pidController->pidReset();
209 dcMotorControl->setSpeedPWM(manualMotorPWM);
210 }
211 } else {
212 pidController->pidReset();
213 dcMotorControl->setSpeedPWM(0);
214 }
215}

217void BowControl::updateMotorStatus() {
218 if (dcMotorControl->getSpeedPWM() != 0) {
219 if (dcMotorControl->isOverPower() || dcMotorControl->isOverCurrent()) {
220 debugPrintln("Bow over power!", debugPrintType::Error);
221 commands->addCommands(commandsOverPowerCurrent);
222 }
223 if (dcMotorControl->getMotorFault()) {
224 debugPrintln("Bow motor fault!", debugPrintType::Error);
225 commands->addCommands(commandsMotorFault);
226 }
227 }

230void BowControl::update() {
231 uint32_t currentTime = micros();
232
233 if (currentTime - lastUpdate >= pidUpdateInterval) {
234 lastUpdate = currentTime;
235
236 updateMotorAutoShutdown();
237 updateRun_PID();
238 updateMotorStatus();
239 bowPressure->update();
240 // Check if the bowing wheel speed is zero
241 dcMotorControl->checkTachometerTimeout();
242 }
243
244 bowPressure->update();
245}
