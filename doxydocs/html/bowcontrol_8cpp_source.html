120void bowControl::pidControl() {
121 // Calculate the error between the target speed and the current speed
122 float currentSpeed = bowIOConnect->getLastTachoFreq();
123 float error = pidTargetSpeed - currentSpeed;
124
125 if (abs(error) > pidPeakError) { pidPeakError = abs(error); }
126
127 if (error > pidMaxError) { error = pidMaxError; }
128 if (error < -pidMaxError) { error = -pidMaxError; }
129
130 // Update the integral term
131 if ((error >= integratorIgnoreBelow) || (error <= -integratorIgnoreBelow)) { integral += error; }
132
133 // Calculate the PID control output
134 KpTerm = Kp * error;
135 KiTerm = Ki * integral;
136 KdTerm = Kd * (error - previousError);
137
138 // At incorrect Ki values the integral starts building,this doesn't really matter since
139 // the type cast makes the output wrap but it might cause trouble in the future so added this
140 if (KiTerm > 65535) { integral -= 65535;}
141
142 float output = KpTerm + KiTerm + KdTerm;
143
144 if (output < 0) { output = 0; }
145 // This attempt at setting lowest speed doesn't work, makes you not be able to go beyond a certain key
146/* if ((pidTargetSpeed > 0) && (output < calibrationDataConnect->minInertialPWM)) {
147 output = calibrationDataConnect->minInertialPWM;
148 }
149*/
150 // If problem, check the following line - added 2023-10-22
151 if (output > 65535) { output = 65535; }
152
153 // Set the motor speed using PWM
154 bowIOConnect->setSpeedPWM(static_cast<uint16_t>(output));
155
156 // Store the current error for the next iteration
157 previousError = error;
158};

185void bowControl::setBowPressureSafe(uint16_t tilt) {
186 if (tilt > calibrationDataConnect->stallPressure) { tilt = calibrationDataConnect->stallPressure; }
187
188 if ((tiltMode == Engage) && (bowIOConnect->stepServoStepper->reachedTarget)) {
189 reachedEngage = true;
190 if (outputDebugData) { debugPrintln("Reached engage", debugPrintType::Debug); }
191 }
192
193 if ((tiltMode == Engage) && (reachedEngage)) {
194 bowIOConnect->stepServoStepper->setSpeed(bowSpeedWhileEngaged);
195 if (outputDebugData) { debugPrintln("Setting pressure speed to slow", debugPrintType::Debug); }
196 } else {
197 bowIOConnect->stepServoStepper->setSpeed(bowSpeedToEngage);
198 if (outputDebugData) { debugPrintln("Setting pressure speed to high", debugPrintType::Debug); }
199 }
200
201 bowIOConnect->setTiltPWM(tilt);
202 return;
203}

428void bowControl::measureTimeToTarget(float _pidTargetSpeed) {
429 elapsedMillis timeToTarget;
430 elapsedMicros overshootTime;
431 float startSpeed = getPIDTarget();
432
433 timeToTarget = 0;
434 float overshoot = 0;
435 float undershoot = -1;
436 float freq;
437 int iterations = 0;
438
439 setPIDTarget(_pidTargetSpeed);
440 while(iterations < 10) {
441 if (round(bowIOConnect->averageFreq()) == round(_pidTargetSpeed)) { iterations++; }
442
443 overshootTime = 0;
444 while (overshootTime < 10) {
445 freq = bowIOConnect->getLastTachoFreq();
446 if (freq > overshoot) { overshoot = freq; }
447 if ((freq < undershoot) || (undershoot = -1 )) { undershoot = freq; }
448 }
449 if (timeToTarget > 1000) {
450 debugPrintln("Couldn't get to target speed in time (reached " + String(bowIOConnect->averageFreq()) + " Hertz)", Error);
451 return;
452 }
453 }
454
455 debugPrint("Going from " + String(startSpeed) + " -> " + String(_pidTargetSpeed) + " in " + String(timeToTarget) + "ms", InfoRequest);
456 if (_pidTargetSpeed > startSpeed) { debugPrintln(", overshoot " + String(overshoot), InfoRequest); }
457 if (_pidTargetSpeed < startSpeed) { debugPrintln(", undershoot " + String(undershoot), InfoRequest); }
458 //, max freq " + String(overshoot) + " Hz, min freq " + String(undershoot) + "Hz", InfoRequest);
459}

477void bowControl::updateString() {
478 unsigned long currentTime = micros();
479
480 bowIOConnect->updateBow();
481
482 // Process only if at the pidUpdateInterval
483 if (currentTime - previousTime >= pidUpdateInterval) {
484 previousTime = currentTime;
485
486 // Calculate the manual Tilt PWM
487 //unsigned int tiltPWM = calibrationDataConnect->minUsablePressure - 8000 +
488 // ((double) (calibrationDataConnect->maxUsablePressure - calibrationDataConnect->minUsablePressure + 16000) / 65535 * manualTiltPWM);
489
491/* if (checkMotorFault()) {
492 debugPrintln("Motor driver Fault!", Error);
493 }*/
494
495 if ((speedMode == Automatic) && (_hold == false) && (bowShutoffTimer >= bowShutoffTimeout) ) {
496 if ((tiltMode == Rest) && (bowShutoffTimedout == false)) {
497 run = 0;
498 bowShutoffTimedout = true;
499 if (outputDebugData) { debugPrintln("Auto shutdown of motor", Debug); }
500 } else
501 if ((bowShutoffTimedout == true) && (bowShutoffMotorDisabled == false) && (bowIOConnect->averageFreq() == 0)) {
502 bowIOConnect->disableBowPower();
503 bowShutoffMotorDisabled = true;
504 if (outputDebugData) { debugPrintln("Auto shutdown of motor dc/dc converter", Debug); }
505 }
506 }
507
508 if (run) {
509 if (PIDon) {
510 if (pidTargetSpeed == 0) {
511 bowIOConnect->setSpeedPWM(0);
512 }
513 } else {
514 pidReset();
515 bowIOConnect->setSpeedPWM(manualSpeedPWM);
516 }
517 } else {
518 // If the string module is not running, turn off the bowing wheel
519 pidReset();
520 bowIOConnect->setSpeedPWM(0);
521 }
522
523 // Check if the bowing wheel speed is zero
524 bowIOConnect->checkTimeout();
525
526 // Check if bow has been put into mute and been at the mute position for long enough
527 if (((tiltMode == Mute) && (mutePeriod > 0)) && (elapsedSinceMute >= mutePeriod)) {
528 bowRest(1);
529 }
530 }
