41controlReader::controlReader(uint8_t inDataReadyPin, uint8_t inGatePin)
42{
43// ADC[0].init(1); // = new ADC(1);
44// ADC[1].init(0); // = new ADC(0);
45
46 setDefaults();
47
48 pinDataReady = inDataReadyPin;
49 pinGate = inGatePin;
50
51 pinMode(pinDataReady, INPUT);
52 pinMode(pinGate, INPUT);
53
54 averages[4].trigger = true;
55 averages[4].dataAverageLength = 1;
56 averages[5].trigger = true;
57
58 resetAds();
59
60 attachInterrupt(digitalPinToInterrupt(pinDataReady), IRS_AdsDataReady, FALLING);
61 attachInterrupt(digitalPinToInterrupt(pinGate), IRS_AdsDataReady2, FALLING);
62 return;
63}

68void controlReader::resetAds() {
69 adsInit = false;
70
71 if (!ads.begin(0x48, &Wire)) {
72 debugPrintln("Failed to initialize ADS.", debugPrintType::Error);
73 return;
74 } else {
75 ads.setGain(GAIN_ONE); //GAIN_TWOTHIRDS
76 ads.startADCReading(ADS1X15_REG_CONFIG_MUX_SINGLE_0, false);
77 ads.setDataRate(RATE_ADS1115_860SPS);
78 }
79
80 if (!ads2.begin(0x49, &Wire)) {
81 debugPrintln("Failed to initialize ADS2", debugPrintType::Error);
82 return;
83 }
84
85 adsInit = true;
86 ads2.setGain(GAIN_ONE);
87 ads2.setDataRate(RATE_ADS1015_3300SPS);
88
89 Wire.setClock(400000);
90
91 ads2.startADCReading(ADS1X15_REG_CONFIG_MUX_SINGLE_0, false);
92
93 currentChannel = ADS1X15_REG_CONFIG_MUX_SINGLE_0;
94 currentChannel2 = ADS1X15_REG_CONFIG_MUX_SINGLE_0;
95
96 debugPrintln("ADS Initialized", debugPrintType::Debug);
97}

117 if ((adsErrorReported || ads2ErrorReported) && ((millis() > (adsReinitCountStart + adsReinitializeTimeout)))) {
118 adsReinitCountStart = millis();

122
123 if (!adsInit) { return; }
124
125 if (adsNewData2) {

129
130 uint8_t ch = 0;
131 switch(currentChannel2) {
132 case ADS1X15_REG_CONFIG_MUX_SINGLE_0:

138 currentChannel2 = ADS1X15_REG_CONFIG_MUX_SINGLE_2; // Stop here

142 currentChannel2 = ADS1X15_REG_CONFIG_MUX_SINGLE_3;

189 ads2ErrorReported = false;
190 } else {
191 if ((millis() > (ads2ConversionStart + ads2TimeOut)) && (!ads2ErrorReported)) {
192 debugPrintln("No data on ADS2!", Error);

228 }
