53eProcessResult ControlReader::processSerialCommand(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request, bool delegate, commandList *delegatedCommands) {
54
55 processCommandItems(inCommandItem, serialCommandsControlReader, sizeof(serialCommandsControlReader) / sizeof(serialCommandItem));
56
57 eProcessResult processResult = eProcessResult::Ok;
58
59 if (inCommandItem->command == "help") {
60 addCommandHelp(serialCommandsControlReader, sizeof(serialCommandsControlReader) / sizeof(serialCommandItem), commandResponses,"");
61 processResult = eProcessResult::PassThrough;
62 } else
63 if (inCommandItem->command == "adcread") {
64 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
65 int i = inCommandItem->argument[0].toInt();
66 if ((i < 0) || (i > 7)) { return eProcessResult::WrongArgumentValue; }
67
68 int32_t convertedValue;
69 if (i < 5) {
70 convertedValue = (int32_t) ((float) getData(i) * ((float) 65536 / 32767));
71 } else {
72 convertedValue = (int32_t) ((float) getData(i) * ((float) 65536 / 2048));
73 }
74 if (convertedValue > 65535) { convertedValue = 65535; }
75 commandResponses->push_back({ "adcr:" + String(i) + ":" + String(convertedValue) + ":" + String(getData(i)), debugPrintType::InfoRequest});
76 } else
77 if (inCommandItem->command == "adccommandmap") {
78 if (!checkArgumentsMin(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
79 uint8_t channel = inCommandItem->argument[0].toInt();
80
81 if (request) {
82 commandResponses->push_back({ "acm:" + String(channel) + ":" + delimitExpression(cvInputCommands[channel], true), InfoRequest});
83 } else {
84 if (!checkArguments(inCommandItem, commandResponses, 2)) { return eProcessResult::WrongArgumentCount; }
85 setADCCommands(channel, stripQuotes(inCommandItem->argument[1]));
86 commandResponses->push_back({ "acm:" + String(channel) + ":" + delimitExpression(cvInputCommands[channel], true), InfoRequest});
87 }
88 } else
89 if (inCommandItem->command == "adcdefaults") {
90 setDefaults();
91 commandResponses->push_back({ "acd:1", debugPrintType::InfoRequest});
92 } else
93 if (inCommandItem->command == "adcsettings") {
94 if (!checkArgumentsMin(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
95 uint8_t channel = inCommandItem->argument[0].toInt();
96 if (channel > 7) { return eProcessResult::WrongArgumentValue; }
97
98 if (request) {
99 commandResponses->push_back({ "adcs:" + String(channel) + ":" + getADCAveragerSettings(channel), InfoRequest});
100 } else {
101 if (!checkArguments(inCommandItem, commandResponses, 5)) { return eProcessResult::WrongArgumentCount; }
102 if (!setADCAveragerSettings(inCommandItem->argument[0].toInt(), inCommandItem->argument[1].toInt(), inCommandItem->argument[2].toInt(), inCommandItem->argument[3].toInt(),
103 inCommandItem->argument[4].toInt())) { return eProcessResult::CommandFailed; }
104 commandResponses->push_back({ "adcs:" + String(channel) + ":" + getADCAveragerSettings(channel), debugPrintType::InfoRequest});
105 }
106 } else {
107 return eProcessResult::NotFound;
108 }
109
110 return processResult;
111}

113eProcessResult ControlReader::processSerialCommandHidden(commandItem *inCommandItem, std::vector<commandResponse> *commandResponses, bool request, bool delegate, commandList *delegatedCommands) {
114
115 processCommandItems(inCommandItem, serialCommandsControlReaderHidden, sizeof(serialCommandsControlReaderHidden) / sizeof(serialCommandItem));
116
117 eProcessResult processResult = eProcessResult::Ok;
118
119 if (inCommandItem->command == "helphidden") {
120 addCommandHelp(serialCommandsControlReaderHidden, sizeof(serialCommandsControlReaderHidden) / sizeof(serialCommandItem), commandResponses,"");
121 processResult = eProcessResult::PassThrough;
122 } else
123 if (inCommandItem->command == "testadclatency") {
124 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
125 testMeasurement = 0;
126 testMeasurementOngoing = true;
127 analogWrite(3, int(inCommandItem->argument[0].toInt()));
128 commandResponses->push_back({"Starting ADC latency test", InfoRequest});
129 } else
130 if (inCommandItem->command == "testadclatencyreturn") {
131 testMeasurementOngoing = false;
132 commandResponses->push_back({"Returned from ADC latency test in " + String(testMeasurement) + " uS", InfoRequest});
133 } else
134 if (inCommandItem->command == "testadcminmax") {
135 if (request) {
136 commandResponses->push_back({ "tamm:" + String(testChannel) + ":" + String(testMin) + ":" + String(testMax), InfoRequest });
137 } else {
138 if (!checkArguments(inCommandItem, commandResponses, 1)) { return eProcessResult::WrongArgumentCount; }
139 int testChannel = inCommandItem->argument[0].toInt();
140 if ((testChannel < 0) || (testChannel > 7)) { return eProcessResult::WrongArgumentValue; }
141 setADCMinMaxTestChannel(testChannel);
142 commandResponses->push_back({ "Starting ADC min/max test on channel " + String(testChannel), InfoRequest });
143 }
144 } else {
145 return eProcessResult::NotFound;
146 }
147
148 return processResult;

154void ControlReader::resetAds() {
155 adsInit = false;
156
157 if (!ads.begin(0x48, &Wire)) {
158 debugPrintln("Failed to initialize ADS.", debugPrintType::Error);
159 return;
160 } else {
161 ads.setGain(GAIN_ONE); //GAIN_TWOTHIRDS
162 ads.startADCReading(ADS1X15_REG_CONFIG_MUX_SINGLE_0, false);
163 ads.setDataRate(RATE_ADS1115_860SPS);
164 }
165
166 if (!ads2.begin(0x49, &Wire)) {
167 debugPrintln("Failed to initialize ADS2", debugPrintType::Error);
168 return;
169 }
170
171 adsInit = true;
172 ads2.setGain(GAIN_ONE);
173 ads2.setDataRate(RATE_ADS1015_3300SPS);
174
175 Wire.setClock(400000);
176
177 ads2.startADCReading(ADS1X15_REG_CONFIG_MUX_SINGLE_0, false);
178
179 currentChannel = ADS1X15_REG_CONFIG_MUX_SINGLE_0;
180 currentChannel2 = ADS1X15_REG_CONFIG_MUX_SINGLE_0;
181
182 debugPrintln("ADS Initialized", debugPrintType::Debug);
183}

185bool ControlReader::readSingleADS(Adafruit_ADS1X15 &adsx, volatile bool &newData, uint16_t &channel, uint8_t channelOffset, long &conversionStart, bool &errorReported, uint16_t timeout, uint16_t resolution) {
186 int16_t a;
187
188 if (newData) {
189 if (adsx.conversionComplete()) {
190 newData = false;
191 a = adsx.getLastConversionResults();
192 uint8_t ch = 0;
193 switch(channel) {
194 case ADS1X15_REG_CONFIG_MUX_SINGLE_0:
195 ch = 0;
196 channel = ADS1X15_REG_CONFIG_MUX_SINGLE_1;
197 break;
198 case ADS1X15_REG_CONFIG_MUX_SINGLE_1:
199 ch = 1;
200 channel = ADS1X15_REG_CONFIG_MUX_SINGLE_2; // Stop here
201 break;
202 case ADS1X15_REG_CONFIG_MUX_SINGLE_2:
203 ch = 2;
204 channel = ADS1X15_REG_CONFIG_MUX_SINGLE_3;
205 break;
206 case ADS1X15_REG_CONFIG_MUX_SINGLE_3:
207 ch = 3;
208 channel = ADS1X15_REG_CONFIG_MUX_SINGLE_0;
209 break;
210 }
211
212 if (a < 0) { a = 0; }
213
214
215 if (testMeasurementOngoing) {
216 debugPrintln("New data (" + String(a) + ") on ADC channel " + String(ch + channelOffset) + " at " + String(testMeasurement) + "uS", Debug);
217 }
218 if (testChannel == (ch + channelOffset)) {
219 addTestData(a);
220 }
221
222 averages[channelOffset + ch].addData(a);
223 if (averages[channelOffset + ch].dataChanged()) {
224 // Convert value to 0-65535 range
225 int32_t convertedValue = (int32_t) ((float) averages[channelOffset + ch].value * ((float) 65536 / resolution));
226 if (convertedValue > 65535) { convertedValue = 65535; }
227 dvalue = (double) convertedValue;
228 if (outputDebugData) {
229 debugPrintln("adcr:" + String(ch + channelOffset) + ":" + String(convertedValue) + ":" + String(averages[channelOffset + ch].value), debugPrintType::InfoRequest);
230 }
231
232 if (!testMeasurementOngoing) {
233 processLocalMessage(&cvInputCommands[channelOffset + ch]);
234 } else {
235 String tempMessages = "talr";
236 processLocalMessage(&tempMessages);
237 }
238
239 // The process adds roughly 5mS of latency, add to queue as priority aka first in line?
240 }
241
242 adsx.startADCReading(channel, false);
243 conversionStart = millis();
244 errorReported = false;
245 } else {
246 if ((millis() > (conversionStart + timeout)) && (!errorReported)) {
247 debugPrintln("No data on ADSx!", Error);
248 errorReported = true;
249 adsReinitCountStart = millis();
250 return false;
251 }
252 }
253 }
254
255 return true;
256}

258void ControlReader::readData() {
259 if ((adsErrorReported || ads2ErrorReported) && ((millis() > (adsReinitCountStart + adsReinitializeTimeout)))) {
260 adsReinitCountStart = millis();
261 debugPrintln("Trying to reinitialize control box", debugPrintType::Debug);
262 resetAds();
263 }
264
265 if (!adsInit) { return; }
266
267 readSingleADS(ads2, adsNewData2, currentChannel2, 4, ads2ConversionStart, ads2ErrorReported, ads2TimeOut, 2048);
268 readSingleADS(ads, adsNewData, currentChannel, 0, adsConversionStart, adsErrorReported, adsTimeOut, 32767);
269}
